\begin{proposition}[forthel]
  Let $n, m, k$ be natural numbers such that $n, m \NotEq \Zero$ and $k \Greater \One$.
  Then $\Exp{k}{n} \Divides \Exp{k}{m}$ iff $n \Leq m$.
\end{proposition}
\begin{proof}[forthel]
  \begin{case}{$\Exp{k}{n} \Divides \Exp{k}{m}$.}
    Assume $n \Greater m$.
    Take a nonzero natural number $l$ such that $n \Eq m \Plus l$.
    Then $\Exp{k}{n}
      \Eq \Exp{k}{m \Plus l}
      \Eq \Exp{k}{m} \Mul \Exp{k}{l}$.
    Hence $\Exp{k}{m} \Divides \Exp{k}{n}$.
    Thus $\Exp{k}{m} \Eq \Exp{k}{n}$.
    Therefore $m \Eq n$.
    Contradiction.
  \end{case}

  \begin{case}{$n \Leq m$.}
    Take a natural number $l$ such that $m \Eq n \Plus l$.
    Then $\Exp{k}{m}
      \Eq \Exp{k}{n \Plus l}
      \Eq \Exp{k}{n} \Mul \Exp{k}{l}$.
    Hence $\Exp{k}{n} \Divides \Exp{k}{m}$.
  \end{case}
\end{proof}

\begin{proposition}[forthel]
  Let $n$ be a composite natural number.
  Then $n$ has a nontrivial divisor $m$ such that $\Exp{m}{\Two} \Leq n$.
\end{proposition}
\begin{proof}[forthel]
  Define $A \DefEq \SClass{m}{\Naturals}{m\text{ is a nontrivial divisor of }n}$.
  $A$ is nonempty.
  Hence we can take a $m \Elem A$ such that $m \Leq l$ for all $l \Elem A$.
  Consider a natural number $k$ such that $m \Mul k \Eq n$.
  Then $m \Leq k$.
  Hence $\Exp{m}{\Two} \Eq m \Mul m \Leq m \Mul k \Eq n$.
  Therefore $\Exp{m}{\Two} \Leq n$.
\end{proof}
