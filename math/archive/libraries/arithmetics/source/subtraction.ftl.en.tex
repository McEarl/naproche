\begin{smodule}{subtraction.ftl}
\importmodule[libraries/arithmetics]{ordering.ftl}
\symdef{Minus}[name=difference]{\,-\,}

\begin{definition}[forthel,for=difference]
  Let $n, m$ be natural numbers such that $n \Geq m$.
  $\emph{n \Minus m}$ is the natural number $k$ such that $n \Eq m \Plus k$.
  Let the \emph{difference of $n$ and $m$} stand for $n \Minus m$.
\end{definition}

\begin{proposition}[forthel]
  Let $n, m$ be natural numbers such that $n \Geq m$.
  Then $n \Minus m \Eq \Zero$ iff $n \Eq m$.
\end{proposition}
\begin{proof}[forthel]
  \begin{case}{$n \Minus m \Eq \Zero$.}
    Then $n
      \Eq (n \Minus m) \Plus m
      \Eq \Zero \Plus m
      \Eq m$.
  \end{case}

  \begin{case}{$n \Eq m$.}
    We have $(n \Minus m) \Plus m
      \Eq n
      \Eq m
      \Eq \Zero \Plus m$.
    Hence $n \Minus m \Eq \Zero$.
  \end{case}
\end{proof}

\begin{corollary}[forthel]
  Let $n$ be a natural number.
  Then $n \Minus n \Eq \Zero$.
\end{corollary}

\begin{proposition}[forthel]
  Let $n$ be a natural number.
  Then $n \Minus \Zero \Eq n$.
\end{proposition}
\begin{proof}[forthel]
  We have $n
    \Eq (n \Minus \Zero) \Plus \Zero
    \Eq n \Minus \Zero$.
\end{proof}

\begin{proposition}[forthel]
  Let $n, m$ be natural numbers such that $n \Geq m$.
  Then $n \Minus m \Leq n$.
\end{proposition}
\begin{proof}[forthel]
  We have $(n \Minus m) \Plus m \Eq n$.
  Hence $n \Minus m \Leq n$.
\end{proof}

\begin{proposition}[forthel]
  Let $n, m$ be natural numbers such that $n \Geq m$.
  If $\Zero \NotEq m \Less n$ then $n \Minus m \Less n$.
\end{proposition}
\begin{proof}[forthel]
  Assume $\Zero \NotEq m \Less n$.
  Suppose $n \Minus m \Geq n$.
  We have $(n \Minus m) \Plus m \Eq n$.
  Then $n \Plus m
    \Eq (n \Minus m) \Plus m
    \Eq n
    \Eq n \Plus \Zero$.
  Hence $m \Eq \Zero$.
  Contradiction.
\end{proof}

\begin{proposition}[forthel]
  Let $n, m, k$ be natural numbers such that $n \Geq m$.
  Then $(n \Minus m) \Plus k \Eq (n \Plus k) \Minus m$.
\end{proposition}
\begin{proof}[forthel]
  We have
  \[  ((n \Minus m) \Plus k) \Plus m       \]
  \[    \Eq ((n \Minus m) \Plus m) \Plus k   \]
  \[    \Eq n \Plus k               \]
  \[    \Eq ((n \Plus k) \Minus m) \Plus m.  \]

  Hence $(n \Minus m) \Plus k \Eq (n \Plus k) \Minus m$.
\end{proof}

\symdecl*{associativity of addition and subtraction}

\begin{corollary}[forthel,name=associativity of addition and subtraction]
  Let $n, m, k$ be natural numbers such that $n \Geq m$.
  Then $k \Plus (n \Minus m) \Eq (k \Plus n) \Minus m$.
\end{corollary}

\begin{proposition}[forthel]
  Let $n, m, k$ be natural numbers such that $n \Geq  m \Plus k$.
  Then $(n \Minus m) \Minus k \Eq n \Minus (m \Plus k)$.
\end{proposition}
\begin{proof}[forthel]
  We have
  \[  ((n \Minus m) \Minus k) \Plus (m \Plus k)       \]
  \[    \Eq (((n \Minus m) \Minus k) \Plus k) \Plus m   \]
  \[    \Eq (n \Minus m) \Plus m               \]
  \[    \Eq n                         \]
  \[    \Eq (n \Minus (m \Plus k)) \Plus (m \Plus k).  \]

  Hence $(n \Minus m) \Minus k \Eq n \Minus (m \Plus k)$.
\end{proof}
