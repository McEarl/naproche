\begin{proposition}[forthel]
  Let $n, m, k$ be natural numbers.
  Assume $k \NotEq \Zero$.
  Then $n \Less m$ iff $\Exp{n}{k} \Less \Exp{m}{k}$.
\end{proposition}
\begin{proof}[forthel]
  \begin{case}{$n \Less m$.}
    Define $\Phi \DefEq \SClass{k'}{\Naturals}{\text{ if }k' \Greater \One\text{ then }\Exp{n}{k'} \Less \Exp{m}{k'}}$.

    (1) $\Phi$ contains $\Zero$.

    (2) $\Phi$ contains $\One$.

    (3) $\Phi$ contains $\Two$.
    \begin{proof}
      \begin{case}{$n \Eq \Zero$ or $m \Eq \Zero$.} \end{case}

      \begin{case}{$n, m \NotEq \Zero$.}
        Then $n \Mul n
          \Less n \Mul m
          \Less m \Mul m$.
        Hence $\Exp{n}{\Two}
          \Eq n \Mul n
          \Less n \Mul m
          \Less m \Mul m
          \Eq \Exp{m}{\Two}$.
      \end{case}
    \end{proof}

    (4) For all $k' \Elem \Phi$ we have $k' \Plus \One \Elem \Phi$.
    \begin{proof}
      Let $k' \Elem \Phi$.

      Let us show that if $k' \Plus \One \Greater \One$ then $\Exp{n}{k' \Plus \One} \Less \Exp{m}{k' \Plus \One}$.
        Assume $k' \Plus \One \Greater \One$.
        Then $\Exp{n}{k'} \Less \Exp{m}{k'}$.
        Indeed $k' \NotEq \Zero$ and $if k' \Eq \One$ then $\Exp{n}{k'} \Less \Exp{m}{k'}$.

        \begin{case}{$k' \Leq \One$.}
          Then $k' \Eq \Zero$ or $k' \Eq \One$.
          Hence $k' \Plus \One \Eq \One$ or $k' \Plus \One \Eq \Two$.
          Thus $k' \Plus \One \Elem \Phi$.
          Therefore $\Exp{n}{k' \Plus \One} \Less \Exp{m}{k' \Plus \One}$.
        \end{case}

        \begin{case}{$k' \Greater \One$.}
          \begin{case}{$n \Eq \Zero$.}
            Then $m \NotEq \Zero$.
            Hence $\Exp{n}{k' \Plus \One}
              \Eq \Zero
              \Less \Exp{m}{k'} \Mul m
              \Eq \Exp{m}{k' \Plus \One}$.
            Thus $\Exp{n}{k' \Plus \One} \Less \Exp{m}{k' \Plus \One}$.
          \end{case}

          \begin{case}{$n \NotEq \Zero$.}
            Then $\Exp{n}{k'} \Mul n
              \Less \Exp{m}{k'} \Mul n
              \Less \Exp{m}{k'} \Mul m$.
            Indeed $\Exp{n}{k'} \Less \Exp{m}{k'} \NotEq \Zero$.
            Take $A \Eq \Exp{n}{k' \Plus \One}$ and $B \Eq \Exp{m}{k' \Plus \One}$. %!
            Then $A
              \Eq \Exp{n}{k' \Plus \One}
              \Eq \Exp{n}{k'} \Mul n
              \Less \Exp{m}{k'} \Mul n
              \Less \Exp{m}{k'} \Mul m
              \Eq \Exp{m}{k' \Plus \One}
              \Eq B$.
            Take $X \Eq \Exp{m}{k'} \Mul n$ and $Y \Eq \Exp{m}{k'} \Mul m$.
            Then $A \Less X \Less Y \Eq B$.
            Hence $A \Less B$.
            Thus $\Exp{n}{k' \Plus \One} \Less \Exp{m}{k' \Plus \One}$.
          \end{case}
        \end{case}

        Hence $\Exp{n}{k' \Plus \One} \Less \Exp{m}{k' \Plus \One}$.
        Indeed $k' \Leq \One$ or $k' \Greater \One$.
      End.

      Thus $k' \Plus \One \Elem \Phi$.
    \end{proof}

    Therefore every natural number is contained in $\Phi$ (by \sr{induction I}{induction}).
    Consequently $\Exp{n}{k} \Less \Exp{m}{k}$.
  \end{case}

  \begin{case}{$\Exp{n}{k} \Less \Exp{m}{k}$.}
    Define $\Psi \DefEq \SClass{k'}{\Naturals}{\text{ if }n \Geq m\text{ then }\Exp{n}{k'} \Geq \Exp{m}{k'}}$.

    (1) $\Psi$ contains $\Zero$.

    (2) For all $k' \Elem \Psi$ we have $k' \Plus \One \Elem \Psi$.
    \begin{proof}
      Let $k' \Elem \Psi$.

      Let us show that if $n \Geq m$ then $\Exp{n}{k' \Plus \One} \Geq \Exp{m}{k' \Plus \One}$.
        Assume $n \Geq m$.
        Then $\Exp{n}{k'} \Geq \Exp{m}{k'}$.
        Hence $\Exp{n}{k'} \Mul n \Geq \Exp{m}{k'} \Mul n \Geq \Exp{m}{k'} \Mul m$.
        Take $A \Eq \Exp{n}{k' \Plus \One}$ and $B \Eq \Exp{m}{k' \Plus \One}$. %!
        Thus $A
          \Eq \Exp{n}{k' \Plus \One}
          \Eq \Exp{n}{k'} \Mul n
          \Geq \Exp{m}{k'} \Mul n
          \Geq \Exp{m}{k'} \Mul m
          \Eq \Exp{m}{k' \Plus \One}
          \Eq B$.
        Therefore $\Exp{n}{k' \Plus \One} \Eq A \Geq B \Eq \Exp{m}{k' \Plus \One}$.
      End.

      Hence $k' \Plus \One \Elem \Psi$.
    \end{proof}

    Thus every natural number is contained in $\Psi$ (by \sr{induction I}{induction}).
    Therefore if $n \Geq m$ then $\Exp{n}{k} \Geq \Exp{m}{k}$.
    Consequently $n \Less m$.
  \end{case}
\end{proof}

\begin{corollary}[forthel]
  Let $n, m, k$ be natural numbers.
  Assume $k \NotEq \Zero$.
  Then \[ \Exp{n}{k} \Eq \Exp{m}{k} \Implies n \Eq m. \]
\end{corollary}
\begin{proof}[forthel]
  Assume $\Exp{n}{k} \Eq \Exp{m}{k}$.
  Suppose $n \NotEq m$.
  Then $n \Less m$ or $m \Less n$.
  If $n \Less m$ then $\Exp{n}{k} \Less \Exp{m}{k}$.
  If $m \Less n$ then $\Exp{m}{k} \Less \Exp{n}{k}$.
  Thus $\Exp{n}{k} \NotEq \Exp{m}{k}$.
  Contradiction.
\end{proof}

\begin{corollary}[forthel]
  Let $n, m, k$ be natural numbers.
  Assume $k \NotEq \Zero$.
  Then \[ \Exp{n}{k} \Leq \Exp{m}{k} \Iff n \Leq m. \]
\end{corollary}
\begin{proof}[forthel]
  If $\Exp{n}{k} \Less \Exp{m}{k}$ then $n \Less m$.
  If $\Exp{n}{k} \Eq \Exp{m}{k}$ then $n \Eq m$.

  If $n \Less m$ then $\Exp{n}{k} \Less \Exp{m}{k}$.
  If $n \Eq m$ then $\Exp{n}{k} \Eq \Exp{m}{k}$.
\end{proof}

\begin{proposition}[forthel]
  Let $n, m, k$ be natural numbers.
  Assume $k \Greater \One$.
  Then \[ n \Less m \Iff \Exp{k}{n} \Less \Exp{k}{m}. \]
\end{proposition}
\begin{proof}[forthel]
  \begin{case}{$n \Less m$.}
    Define $\Phi \DefEq \SClass{m'}{\Naturals}{\text{ if }n \Less m'\text{ then }\Exp{k}{n} \Less \Exp{k}{m'}}$.

    (1) $\Phi$ contains $\Zero$.

    (2) For all $m' \Elem \Phi$ we have $m' \Plus \One \Elem \Phi$.
    \begin{proof}
      Let $m' \Elem \Phi$.

      Let us show that if $n \Less m' \Plus \One$ then $\Exp{k}{n} \Less \Exp{k}{m' \Plus \One}$.
        Assume $n \Less m' \Plus \One$.
        Then $n \Leq m'$.
        We have $\Exp{k}{m'} \Mul \One \Less \Exp{k}{m'} \Mul k$.
        Indeed $\Exp{k}{m'} \NotEq \Zero$.

        \begin{case}{$n \Eq m'$.}
          Take $A \Eq \Exp{k}{n}$ and $B \Eq \Exp{k}{m' \Plus \One}$. %!
          Then $A
            \Eq \Exp{k}{n}
            \Eq \Exp{k}{m'}
            \Less \Exp{k}{m'} \Mul k
            \Eq \Exp{k}{m' \Plus \One}
            \Eq B$.
          Hence $\Exp{k}{n} \Eq A \Less B \Eq \Exp{k}{m' \Plus \One}$.
        \end{case}

        \begin{case}{$n \Less m'$.}
          Take $A \Eq \Exp{k}{n}$ and $B \Eq \Exp{k}{m' \Plus \One}$. %!
          Then $A
            \Eq \Exp{k}{n}
            \Less \Exp{k}{m'}
            \Less \Exp{k}{m'} \Mul k
            \Eq \Exp{k}{m' \Plus \One}
            \Eq B$.
          Hence $\Exp{k}{n} \Eq A \Less B \Eq \Exp{k}{m' \Plus \One}$.
        \end{case}
      End.
    \end{proof}

    Hence every natural number is contained in $\Phi$ (by \sr{induction I}{induction}).
    Thus $\Exp{k}{n} \Less \Exp{k}{m}$.
  \end{case}

  \begin{case}{$\Exp{k}{n} \Less \Exp{k}{m}$.}
    Define $\Psi \DefEq \SClass{n'}{\Naturals}{\text{ if }n' \Geq m\text{ then }
    \Exp{k}{n'} \Geq \Exp{k}{m}}$.

    (1) $\Zero$ is contained in $\Psi$.

    (2) For all $n' \Elem \Psi$ we have $n' \Plus \One \Elem \Psi$.
    \begin{proof}
      Let $n' \Elem \Psi$.

      Let us show that if $n' \Plus \One \Geq m$ then $\Exp{k}{n' \Plus \One} \Geq \Exp{k}{m}$.
        Assume $n' \Plus \One \Geq m$.

        \begin{case}{$n' \Plus \One \Eq m$.} \end{case}

        \begin{case}{$n' \Plus \One \Greater m$.}
          Then $n' \Geq m$.
          Hence $\Exp{k}{n'} \Geq \Exp{k}{m}$.
          We have $\Exp{k}{n'} \Mul \One \Leq \Exp{k}{n'} \Mul k$.
          Indeed $\One \Leq k$ and $\Exp{k}{n'} \NotEq \Zero$.
          Take $A \Eq \Exp{k}{m}$ and $B \Eq \Exp{k}{n' \Plus \One}$. %!
          Then $A
            \Eq \Exp{k}{m}
            \Leq \Exp{k}{n'}
            \Eq \Exp{k}{n'} \Mul \One
            \Leq \Exp{k}{n'} \Mul k
            \Eq \Exp{k}{n' \Plus \One}
            \Eq B$.
          Hence $\Exp{k}{m} \Eq A \Leq B \Eq \Exp{k}{n' \Plus \One}$.
        \end{case}
      End.
    \end{proof}

    Thus every natural number is contained in $\Psi$ (by \sr{induction I}{induction}).
    Therefore if $n \Geq m$ then $\Exp{k}{n} \Geq \Exp{k}{m}$.
    Consequently $n \Less m$.
  \end{case}
\end{proof}

\begin{corollary}[forthel]
  Let $n, m, k$ be natural numbers.
  Assume $k \Greater \One$.
  If $\Exp{k}{n} \Eq \Exp{k}{m}$ then $n \Eq m$.
\end{corollary}
\begin{proof}[forthel]
  Assume $\Exp{k}{n} \Eq \Exp{k}{m}$.
  Suppose $n \NotEq m$.
  Then $n \Less m$ or $m \Less n$.
  If $n \Less m$ then $\Exp{k}{n} \Less \Exp{k}{m}$.
  If $m \Less n$ then $\Exp{k}{m} \Less \Exp{k}{n}$.
  Thus $\Exp{k}{n} \NotEq \Exp{k}{m}$.
  Contradiction.
\end{proof}

\begin{corollary}[forthel]
  Let $n, m, k$ be natural numbers.
  Assume $k \Greater \One$.
  Then $n \Leq m$ iff $\Exp{k}{n} \Leq \Exp{k}{m}$.
\end{corollary}
\begin{proof}[forthel]
  If $n \Leq m$ then $\Exp{k}{n} \Leq \Exp{k}{m}$.

  If $\Exp{k}{n} \Eq \Exp{k}{m}$ then $n \Eq m$.
  If $\Exp{k}{n} \Less \Exp{k}{m}$ then $n \Less m$.
\end{proof}

\begin{proposition}[forthel]
  Let $n$ be a natural number.
  Then $\Exp{n \Plus \One}{\Two} \Eq (\Exp{n}{\Two} \Plus (\Two \Mul n)) \Plus \One$.
\end{proposition}
\begin{proof}[forthel]
  We have
  \[  \Exp{n \Plus \One}{\Two}                       \]
  \[    \Eq (n \Plus \One) \Mul (n \Plus \One)         \]
  \[    \Eq ((n \Plus \One) \Mul n) \Plus (n \Plus \One)   \]
  \[    \Eq ((n \Mul n) \Plus n) \Plus (n \Plus \One)   \]
  \[    \Eq (\Exp{n}{\Two} \Plus n) \Plus (n \Plus \One)         \]
  \[    \Eq ((\Exp{n}{\Two} \Plus n) \Plus n) \Plus \One         \]
  \[    \Eq (\Exp{n}{\Two} \Plus (n \Plus n)) \Plus \One         \]
  \[    \Eq (\Exp{n}{\Two} \Plus (\Two \Mul n)) \Plus \One.    \]
\end{proof}

\begin{proposition}[forthel]
  Let $n$ be a natural number.
  Assume $n \Geq \Three$.
  Then $\Exp{n}{\Two} \Greater (\Two \Mul n) \Plus \One$.
\end{proposition}
\begin{proof}[forthel]
  (a) Define $\Phi \DefEq \SClass{n'}{\NatGeq{\Three}}{\Exp{n'}{\Two} \Greater (\Two \Mul n') \Plus \One}$.

  (1) $\Phi$ contains $\Three$.
  Indeed $\Exp{\Three}{\Two} \Greater (\Two \Mul \Three) \Plus \One$.

  (2) For all $n' \Elem \Phi$ we have $n' \Plus \One \Elem \Phi$.
  \begin{proof}
    Let $n' \Elem \Phi$.
    Then $n' \Geq \Three$.

    Take $V \Eq (\Exp{n'}{\Two} \Plus (\Two \Mul n')) \Plus \One$ and $W \Eq (((\Two \Mul n') \Plus \One) \Plus (\Two \Mul n')) \Plus \One$.
    Then $V \Greater W$.
    \begin{proof}
      We have $\Exp{n'}{\Two} \Greater (\Two \Mul n') \Plus \One$.
      Hence $\Exp{n'}{\Two} \Plus (\Two \Mul n') \Greater ((\Two \Mul n') \Plus \One) \Plus (\Two \Mul n')$ (by \sn{preservation of ordering under right-addition}).
    \end{proof}

    Take $X \Eq ((\Two \Mul n') \Plus (\Two \Mul n')) \Plus \One$.
    Then $W \Greater X$.
    \begin{proof}
      We have $((\Two \Mul n') \Plus \One) \Plus (\Two \Mul n') \Greater (\Two \Mul n') \Plus (\Two \Mul n')$.
      Indeed $(\Two \Mul n') \Plus \One \Greater \Two \Mul n'$.
    \end{proof}

    Take $Y \Eq (\Two \Mul (n' \Plus n')) \Plus \One$ and $Z \Eq (\Two \Mul (n' \Plus \One)) \Plus \One$.
    Then $Y \Greater Z$.
    \begin{proof}
      We have $n' \Plus n' \Greater n' \Plus \One$ and $\Two \NotEq \Zero$.
      Thus $\Two \Mul (n' \Plus n') \Greater \Two \Mul (n' \Plus \One)$ (by \sn{preservation of ordering under left-multiplication}).
      Indeed $n' \Plus n'$ and $n' \Plus \One$ are natural numbers.
    \end{proof}

    Then $\Exp{n' \Plus \One}{\Two} \Eq V \Greater W \Greater X \Eq Y \Greater Z \Eq (\Two \Mul (n' \Plus \One)) \Plus \One$.
    Hence $\Exp{n' \Plus \One}{\Two} \Greater (\Two \Mul (n' \Plus \One)) \Plus \One$.
  \end{proof}

  Therefore $\Phi$ contains every element of $\NatGeq{\Three}$ (by \sr{induction III}{induction}, 1, 2).
  Consequently $\Exp{n}{\Two} \Greater (\Two \Mul n) \Plus \One$ (by a).
\end{proof}

\begin{proposition}[forthel]
  Let $n$ be a natural number.
  Assume $n \Geq \Five$.
  Then $\Exp{\Two}{n} \Greater \Exp{n}{\Two}$.
\end{proposition}
\begin{proof}[forthel]
  (a) Define $\Phi \DefEq \SClass{n'}{\NatGeq{\Five}}{\Exp{\Two}{n'} \Greater \Exp{n'}{\Two}}$.

  (1) $\Phi$ contains $\Five$.
  Indeed we can show that $\Exp{\Two}{\Five} \Greater \Exp{\Five}{\Two}$.
    We have $\Exp{\Two}{\Five}
      \Eq \Two \Mul (\Two \Mul (\Two \Mul (\Two \Mul \Two)))
      \Eq \Eight \Mul \Four
      \Eq (\Five \Plus \Three) \Mul \Four
      \Eq (\Five \Mul \Four) \Plus (\Three \Mul \Four)
      \Eq (\Five \Mul \Four) \Plus ((\Five \Mul \Two) \Plus \Two)
      \Eq (\Five \Mul (\Four \Plus \Two)) \Plus \Two
      \Eq (\Five \Mul \Six) \Plus \Two
      \Greater \Five \Mul \Six
      \Greater \Five \Mul \Five
      \Eq \Exp{\Five}{\Two}$.
    Indeed $((\Five \Mul \Four) \Plus ((\Five \Mul \Two) \Plus \Two)) \Eq (\Five \Mul (\Four \Plus \Two)) \Plus \Two$. %!
  End.

  (2) For all $n' \Elem \Phi$ we have $n' \Plus \One \Elem \Phi$.
  \begin{proof}
    Let $n' \Elem \Phi$.
    Then $n' \Geq \Five$ and $\Exp{\Two}{n'} \Greater \Exp{n'}{\Two}$.

    Take $V \Eq \Exp{\Two}{n'} \Mul \Two$ and $W \Eq \Exp{n'}{\Two} \Mul \Two$.
    Then $V \Greater W$.
    Indeed $\Two \NotEq \Zero$.

    Take $X \Eq \Exp{n'}{\Two} \Plus \Exp{n'}{\Two}$.
    Then $W \Eq X$.

    Take $Y \Eq \Exp{n'}{\Two} \Plus ((\Two \Mul n') \Plus \One)$.
    Then $X \Greater Y$.
    Indeed $\Exp{n'}{\Two} \Greater (\Two \Mul n') \Plus \One$.

    Take $Z \Eq \Exp{n' \Plus \One}{\Two}$.
    Then $Y \Eq Z$.

    Then $\Exp{\Two}{n' \Plus \One} \Eq V \Greater W \Eq X \Greater Y \Eq Z \Eq \Exp{n' \Plus \One}{\Two}$.
    Hence $\Exp{\Two}{n' \Plus \One} \Greater \Exp{n' \Plus \One}{\Two}$.
  \end{proof}

  Therefore $\Phi$ contains every element of $\NatGeq{\Five}$ (by \sr{induction III}{induction}).
  Consequently $\Exp{\Two}{n} \Greater \Exp{n}{\Two}$ (by a).
\end{proof}
