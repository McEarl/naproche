\importmodule[libraries/arithmetics]{natural-numbers.ftl}

\symdef{Mul}[name=product]{\,\cdot\,}

\begin{sfragment}{Definition}
  \begin{signature}[forthel,for=Mul]
    Let $n, m$ be natural numbers.
    $\emph{n \Mul m}$ is a natural number.
    Let the \emph{product of $n$ and $m$} stand for $n \Mul m$.
  \end{signature}

  \begin{axiom}[forthel,for=Mul]
    Let $n$ be a natural number.
    Then $n \Mul \Zero \Eq \Zero$.
  \end{axiom}

  \begin{axiom}[forthel,for=Mul]
    Let $n, m$ be natural numbers.
    Then $n \Mul (m \Plus \One) \Eq (n \Mul m) \Plus n$.
  \end{axiom}
\end{sfragment}

\begin{sfragment}{Computation Laws}
  \begin{sfragment}{Distributivity}

    \symdecl*{left-distributivity of multiplication over addition}

    \begin{proposition}[forthel,name=left-distributivity of multiplication over addition]
      Let $n, m, k$ be natural numbers.
      Then $n \Mul (m \Plus k) \Eq (n \Mul m) \Plus (n \Mul k)$.
    \end{proposition}
    \begin{proof}[forthel]
      Define $\Phi \DefEq \SClass{k'}{\Naturals}{n \Mul (m \Plus k') \Eq (n \Mul m) \Plus (n \Mul k')}$.

      (1) $\Zero$ is an element of $\Phi$.
      Indeed $n \Mul (m \Plus \Zero)
        \Eq n \Mul m
        \Eq (n \Mul m) \Plus \Zero
        \Eq (n \Mul m) \Plus (n \Mul \Zero)$.

      (2) For all $k' \Elem \Phi$ we have $k' \Plus \One \Elem \Phi$. 
      \begin{proof}
        Let $k'\Elem \Phi$.
        Then
        \[  n \Mul (m \Plus (k' \Plus \One))                  \]
        \[    \Eq n \Mul ((m \Plus k') \Plus \One)              \]
        \[    \Eq (n \Mul (m \Plus k')) \Plus n              \]
        \[    \Eq ((n \Mul m) \Plus (n \Mul k')) \Plus n    \]
        \[    \Eq (n \Mul m) \Plus ((n \Mul k') \Plus n)    \]
        \[    \Eq (n \Mul m) \Plus (n \Mul (k' \Plus \One)).   \]
        Hence $n \Mul (m \Plus (k' \Plus \One)) \Eq (n \Mul m) \Plus (n \Mul (k' \Plus \One))$.
        Thus $k' \Plus \One \Elem \Phi$.
      \end{proof}

      Thus every natural number is contained in $\Phi$.
      Therefore $n \Mul (m \Plus k) \Eq (n \Mul m) \Plus (n \Mul k)$.
    \end{proof}

    \begin{proposition}[forthel]
      Let $n, m, k$ be natural numbers.
      Then $(n \Plus m) \Mul k \Eq (n \Mul k) \Plus (m \Mul k)$.
    \end{proposition}
    \begin{proof}[forthel]
      Define $\Phi \DefEq \SClass{k'}{\Naturals}{(n \Plus m) \Mul k' \Eq (n \Mul k') \Plus (m \Mul k')}$.

      (1) $\Zero$ belongs to $\Phi$.
      Indeed $((n \Plus m) \Mul \Zero)
        \Eq \Zero
        \Eq \Zero \Plus \Zero
        \Eq (n \Mul \Zero) \Plus (m \Mul \Zero)$.

      (2) For all $k' \Elem \Phi$ we have $k' \Plus \One \Elem \Phi$.
      \begin{proof}
        Let $k' \Elem \Phi$.
        Then
        \[  (n \Plus m) \Mul (k' \Plus \One)                        \]
        \[    \Eq ((n \Plus m) \Mul k') \Plus (n \Plus m)              \]
        \[    \Eq ((n \Mul k') \Plus (m \Mul k')) \Plus (n \Plus m)   \]
        \[    \Eq (((n \Mul k') \Plus (m \Mul k')) \Plus n) \Plus m   \]
        \[    \Eq ((n \Mul k') \Plus ((m \Mul k') \Plus n)) \Plus m   \]
        \[    \Eq ((n \Mul k') \Plus (n \Plus (m \Mul k'))) \Plus m   \]
        \[    \Eq (((n \Mul k') \Plus n) \Plus (m \Mul k')) \Plus m   \]
        \[    \Eq ((n \Mul k') \Plus n) \Plus ((m \Mul k') \Plus m)   \]
        \[    \Eq (n \Mul (k' \Plus \One)) \Plus (m \Mul (k' \Plus \One)).  \]
        Thus $(n \Plus m) \Mul (k' \Plus \One) \Eq (n \Mul (k' \Plus \One)) \Plus (m \Mul (k' \Plus \One))$.
      \end{proof}

      Thus every natural number is an element of $\Phi$.
      Therefore $(n \Plus m) \Mul k \Eq (n \Mul k) \Plus (m \Mul k)$.
    \end{proof}
  \end{sfragment}

  \begin{sfragment}{Multiplication with $1$ and $2$}
    \begin{proposition}[forthel]
      Let $n$ be a natural number.
      Then $n \Mul \One \Eq n$.
    \end{proposition}
    \begin{proof}[forthel]
      $n \Mul \One
        \Eq n \Mul (\Zero \Plus \One)
        \Eq (n \Mul \Zero) \Plus n
        \Eq \Zero \Plus n
        \Eq n$.
    \end{proof}

    \begin{corollary}[forthel]
      Let $n$ be a natural number.
      Then $n \Mul \Two \Eq n \Plus n$.
    \end{corollary}
    \begin{proof}[forthel]
      $n \Mul \Two
        \Eq n \Mul (\One \Plus \One)
        \Eq (n \Mul \One) \Plus n
        \Eq n \Plus n$.
    \end{proof}
  \end{sfragment}

  \begin{sfragment}{Associativity}
    \begin{proposition}[forthel]
      Let $n, m, k$ be natural numbers.
      Then $n \Mul (m \Mul k) \Eq (n \Mul m) \Mul k$.
    \end{proposition}
    \begin{proof}[forthel]
      Define $\Phi \DefEq \SClass{k'}{\Naturals}{n \Mul (m \Mul k') \Eq (n \Mul m) \Mul k'}$.

      (1) $\Zero$ is contained in $\Phi$.
      Indeed $n \Mul (m \Mul \Zero)
        \Eq n \Mul \Zero
        \Eq \Zero
        \Eq (n \Mul m) \Mul \Zero$.

      (2) For all $k' \Elem \Phi$ we have $k' \Plus \One \Elem \Phi$.
      \begin{proof}
        Let $k' \Elem \Phi$.
        Then
        \[  n \Mul (m \Mul (k' \Plus \One))                          \]
        \[    \Eq n \Mul ((m \Mul k') \Plus m)                      \]
        \[    \Eq (n \Mul (m \Mul k')) \Plus (n \Mul m)            \]
        \[    \Eq ((n \Mul m) \Mul k') \Plus (n \Mul m)            \]
        \[    \Eq ((n \Mul m) \Mul k') \Plus ((n \Mul m) \Mul \One)  \]
        \[    \Eq (n \Mul m) \Mul (k' \Plus \One).                     \]
      \end{proof}

      Hence every natural number is contained in $\Phi$ (by \sr{induction I}{induction}).
      Thus $n \Mul (m \Mul k) \Eq (n \Mul m) \Mul k$.
    \end{proof}
  \end{sfragment}

  \begin{sfragment}{Commutativity}

    \symdecl*{commutativity of multiplication}

    \begin{proposition}[forthel,name=commutativity of multiplication]
      Let $n, m$ be natural numbers.
      Then \[ n \Mul m \Eq m \Mul n. \]
    \end{proposition}
    \begin{proof}[forthel]
      Define $\Phi \DefEq \SClass{m'}{\Naturals}{n \Mul m' \Eq m' \Mul n}$.

      (1) $\Zero$ is contained in $\Phi$.
      \begin{proof}
        Define $\Psi \DefEq \SClass{n'}{\Naturals}{n' \Mul \Zero \Eq \Zero \Mul n'}$.

        (1a) $\Zero$ is contained in $\Psi$.

        (1b) For all $n' \Elem \Psi$ we have $n' \Plus \One \Elem \Psi$.
        \begin{proof}
          Let $n' \Elem \Psi$.
          Then
          \[ (n' \Plus \One) \Mul \Zero
            \Eq \Zero
            \Eq n' \Mul \Zero
            \Eq \Zero \Mul n'
            \Eq (\Zero \Mul n') \Plus \Zero
            \Eq \Zero \Mul (n' \Plus \One). \]
        \end{proof}

        Hence every natural number is contained in $\Psi$ (by \sr{induction I}{induction}).
        Thus $n \Mul \Zero \Eq \Zero \Mul n$.
      \end{proof}

      (2) $\One$ belongs to $\Phi$.
      \begin{proof}
        Define $\Theta \DefEq \SClass{n'}{\Naturals}{n' \Mul \One \Eq \One \Mul n'}$.

        (2a) $\Zero$ is contained in $\Theta$.

        (2b) For all $n' \Elem \Theta$ we have $n' \Plus \One \Elem \Theta$.
        \begin{proof}
          Let $n' \Elem \Theta$.
          Then
          \[  (n' \Plus \One) \Mul \One        \]
          \[    \Eq (n' \Mul \One) \Plus \One    \]
          \[    \Eq (\One \Mul n') \Plus \One    \]
          \[    \Eq \One \Mul (n' \Plus \One).   \]
        \end{proof}

        Thus every natural number is contained in $\Theta$ (by \sr{induction I}{induction}).
        Therefore $n \Mul \One \Eq \One \Mul n$.
      \end{proof}

      (3) For all $m' \Elem \Phi$ we have $m' \Plus \One \Elem \Phi$.
      \begin{proof}
        Let $m' \Elem \Phi$.
        Then
        \[  n \Mul (m' \Plus \One)                \]
        \[    \Eq (n \Mul m') \Plus (n \Mul \One)  \]
        \[    \Eq (m' \Mul n) \Plus (\One \Mul n)  \]
        \[    \Eq (\One \Mul n) \Plus (m' \Mul n)  \]
        \[    \Eq (\One \Plus m') \Mul n            \]
        \[    \Eq (m' \Plus \One) \Mul n.           \]
        Indeed $((\One \Mul n) \Plus (m' \Mul n)) \Eq (\One \Plus m') \Mul n$. %!
      \end{proof}

      Hence every natural number is contained in $\Phi$ (by \sr{induction I}{induction}).
      Thus $n \Mul m \Eq m \Mul n$.
    \end{proof}
  \end{sfragment}

  \begin{sfragment}{Non-Existence of Zero-Divisors}
    \begin{proposition}[forthel]
      Let $n, m$ be natural numbers such that $n \Mul m \Eq \Zero$.
      Then $n \Eq \Zero$ or $m \Eq \Zero$.
    \end{proposition}
    \begin{proof}[forthel]
      Suppose $n, m \NotEq \Zero$.
      Take natural numbers $n', m'$ such that $n \Eq (n' \Plus \One)$ and $m \Eq (m' \Plus \One)$.
      Then
      \[  \Zero                                     \]
      \[    \Eq n \Mul m                         \]
      \[    \Eq (n' \Plus \One) \Mul (m' \Plus \One)           \]
      \[    \Eq ((n' \Plus \One) \Mul m') \Plus (n' \Plus \One)    \]
      \[    \Eq (((n' \Plus \One) \Mul m') \Plus n') \Plus \One.   \]
      Indeed $(n' \Plus \One) \Mul (m' \Plus \One) \Eq ((n' \Plus \One) \Mul m') \Plus (n' \Plus \One)$.
      Hence $\Zero \Eq k \Plus \One$ for some natural number $k$.
      Contradiction.
    \end{proof}
  \end{sfragment}

  \begin{sfragment}{Cancellation}

    \symdecl*{right-cancellability of multiplication}

    \begin{proposition}[forthel,name=right-cancellability of multiplication]
      Let $n, m, k$ be natural numbers.
      Assume $k \NotEq \Zero$.
      If $n \Mul k \Eq m \Mul k$ then $n \Eq m$.
    \end{proposition}
    \begin{proof}[forthel]
      Define $\Phi \DefEq \SClass{n'}{\Naturals}{\text{ for all }m' \Elem \Naturals\text{ if }n' \Mul k \Eq m' \Mul k\text{ and }k \NotEq \Zero\text{ then }n' \Eq m'}$.

      (1) $\Zero$ is contained in $\Phi$.
      \begin{proof}
        Let $m' \Elem \Naturals$.
        Assume $\Zero \Mul k \Eq m' \Mul k$ and $k \NotEq \Zero$.
        Then $m' \Mul k \Eq \Zero$.
        Hence $m' \Eq \Zero$ or $k \Eq \Zero$.
        Thus $m' \Eq \Zero$.
      \end{proof}

      (2) For all $n' \Elem \Phi$ we have $n' \Plus \One \Elem \Phi$. 
      \begin{proof}
        Let $n' \Elem \Phi$.

        Let us show that for all $m' \Elem \Naturals$ if $(n' \Plus \One) \Mul k \Eq m' \Mul k$ and $k \NotEq \Zero$ then $n' \Plus \One \Eq m'$.
          Let $m' \Elem \Naturals$.
          Assume $(n' \Plus \One) \Mul k \Eq m' \Mul k$ and $k \NotEq \Zero$.

          \begin{case}{$m' \Eq \Zero$.}
            Then $(n' \Plus \One) \Mul k \Eq \Zero$.
            Hence $n' \Plus \One \Eq \Zero$.
            Contradiction.
          \end{case}

          \begin{case}{$m' \NotEq \Zero$.}
            Take a natural number $l$ such that $m' \Eq l \Plus \One$.
            Then $(n' \Plus \One) \Mul k \Eq (l \Plus \One) \Mul k$.
            Hence $(n' \Mul k) \Plus k
              \Eq (n' \Mul k) \Plus (\One \Mul k)
              \Eq (n' \Mul k) \Plus k
              \Eq (l \Plus \One) \Mul k
              \Eq (l \Mul k) \Plus (\One \Mul k)
              \Eq (l \Mul k) \Plus k$.
            Thus $n' \Mul k \Eq l \Mul k$ (by \sn{right-cancellability of addition}).
            Indeed $n' \Mul k$ and $l \Mul k$ are natural numbers.
            Then we have $n' \Eq l$.
            Indeed if $n' \Mul k \Eq l \Mul k$ and $k \NotEq \Zero$ then $n' \Eq l$.
            Therefore $n' \Plus \One \Eq l \Plus \One \Eq m'$.
          \end{case}
        End.

        Hence $n' \Plus \One \Elem \Phi$.
      \end{proof}

      Thus every natural number is contained in $\Phi$ (by \sr{induction I}{induction}).
      Therefore if $n \Mul k \Eq m \Mul k$ then $n \Eq m$.
    \end{proof}

    \symdecl*{left-cancellability of multiplication}

    \begin{corollary}[forthel,name=left-cancellability of multiplication]
      Let $n, m, k$ be natural numbers.
      Assume $k \NotEq \Zero$.
      If $k \Mul n \Eq k \Mul m$ then $n \Eq m$.
    \end{corollary}
    \begin{proof}[forthel]
      Assume $k \Mul n \Eq k \Mul m$.
      We have $k \Mul n \Eq n \Mul k$ and $k \Mul m \Eq m \Mul k$.
      Hence $n \Mul k \Eq m \Mul k$.
      Thus $n \Eq m$ (by \sn{right-cancellability of multiplication}).
    \end{proof}
  \end{sfragment}
\end{sfragment}
