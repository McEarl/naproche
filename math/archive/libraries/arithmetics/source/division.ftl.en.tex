\importmodule[libraries/arithmetics]{divisibility.ftl}

\symdef{Frac}[args=2]{\frac{#1}{#2}}

\symdecl*{quotient}

\begin{definition*}[forthel,for=quotient]
  Let $n$ be a natural number and $m$ be a nonzero divisor of $n$.
  $\emph{\Frac{n}{m}}$ is the natural number $k$ such that $k \Mul m \Eq n$.
  Let the \emph{quotient of $n$ and $m$} stand for $\Frac{n}{m}$.
\end{definition*}

\begin{proposition*}[forthel]
  Let $n$ be a natural number.
  $\Frac{n}{\One} \Eq n$.
\end{proposition*}
\begin{proof}[forthel]
  We have $\Frac{n}{\One}
    \Eq \Frac{n}{\One} \Mul \One
    \Eq n$.
\end{proof}

\begin{proposition*}[forthel]
  Let $n$ be a natural number and $m$ be a nonzero divisor of $n$.
  Then $\Frac{n}{m} \Eq \Zero$ iff $n \Eq \Zero$.
\end{proposition*}
\begin{proof}[forthel]
  \begin{case}{$\Frac{n}{m} \Eq \Zero$.}
    Then $n
      \Eq \Frac{n}{m} \Mul m
      \Eq \Zero \Mul m
      \Eq \Zero$.
  \end{case}

  \begin{case}{$n \Eq \Zero$.}
    Then $\Frac{n}{m} \Mul m
      \Eq n
      \Eq \Zero$.
    Hence $\Frac{n}{m} \Eq \Zero$ or $m \Eq \Zero$.
    $m$ is nonzero.
    Thus $\Frac{n}{m} \Eq \Zero$.
  \end{case}
\end{proof}

\begin{proposition*}[forthel]
  Let $n, m, k$ be natural numbers such that $k$ is nonzero.
  Assume $k \Divides n,m$.
  Then \[\Frac{n \Plus m}{k} \Eq \Frac{n}{k} \Plus \Frac{m}{k}.\]
\end{proposition*}
\begin{proof}[forthel]
  We have $\Frac{n \Plus m}{k} \Mul k \Eq n \Plus m$ and $\Frac{n}{k} \Mul k \Eq n$ and $\Frac{m}{k} \Mul k \Eq m$.
  Hence
  \[  \Frac{n \Plus m}{k} \Mul k
      \Eq \left( \Frac{n}{k} \Mul k \right) \Plus \left( \Frac{m}{k} \Mul k \right)
      \Eq \left( \Frac{n}{k} \Plus \Frac{m}{k} \right) \Mul k. \]
  Thus $\Frac{n \Plus m}{k} \Eq \Frac{n}{k} \Plus \Frac{m}{k}$.
\end{proof}

\begin{proposition*}[forthel]
  Let $n, m$ be natural numbers and $k$ be a nonzero divisor of $m$.
  Then \[\Frac{n \Mul m}{k} \Eq n \Mul \Frac{m}{k}.\]
\end{proposition*}
\begin{proof}[forthel]
  We have $\Frac{n \Mul m}{k} \Mul k \Eq n \Mul m$ and $\Frac{m}{k} \Mul k \Eq m$.
  Hence
  \[  \Frac{n \Mul m}{k} \Mul k
      \Eq n \Mul \left( \Frac{m}{k} \Mul k \right)
      \Eq \left( n \Mul \Frac{m}{k} \right) \Mul k. \]
  Thus $\Frac{n \Mul m}{k} \Eq n \Mul \Frac{m}{k}$.
\end{proof}

\begin{proposition*}[forthel]
  Let $n, m$ be natural numbers and $k$ be a nonzero divisor of $m$.
  Then \[\Frac{n \Mul m}{k} \Eq \Frac{m}{k} \Mul n.\]
\end{proposition*}

\begin{proposition*}[forthel]
  Let $n, m$ be natural numbers and $k$ be a nonzero divisor of $n$.
  Then \[\Frac{n \Mul m}{k} \Eq \Frac{n}{k} \Mul m.\]
\end{proposition*}

\begin{proposition*}[forthel]
  Let $n, m$ be natural numbers and $k$ be a nonzero divisor of $n$.
  Then \[\Frac{n \Mul m}{k} \Eq m \Mul \Frac{n}{k}.\]
\end{proposition*}

\begin{proposition*}[forthel]
  Let $n, k$ be natural numbers such that $k$ be nonzero and $m$ be a nonzero divisor of $n$.
  Then \[\Frac{n \Mul k}{m \Mul k} \Eq \Frac{n}{m}.\]
\end{proposition*}
\begin{proof}[forthel]
  We have $\Frac{n \Mul k}{m \Mul k} \Mul (m \Mul k) \Eq n \Mul k$.
  Hence
  \[  \left( \Frac{n \Mul k}{m \Mul k} \Mul m \right) \Mul k
      \Eq \Frac{n \Mul k}{m \Mul k} \Mul (m \Mul k)
      \Eq n \Mul k. \]
  Thus $\Frac{n \Mul k}{m \Mul k} \Mul m \Eq n$.
  Therefore $\Frac{n}{m} \Eq \Frac{n \Mul k}{m \Mul k}$.
\end{proof}

\begin{proposition*}[forthel]
  Let $n, k$ be natural numbers such that $k$ be nonzero and $m$ be a nonzero divisor of $n$.
  Then \[\Frac{k \Mul n}{k \Mul m} \Eq \Frac{n}{m}.\]
\end{proposition*}
