\importmodule[libraries/arithmetics]{division.ftl}
\importmodule[libraries/arithmetics]{exponentiation.ftl}
\importmodule[libraries/arithmetics]{subtraction.ftl}

\begin{proposition*}[forthel]
  Let $n, m, k$ be natural numbers such that $k$ is nonzero and $n \Geq m$.
  Then \[\Frac{\Exp{k}{n}}{\Exp{k}{m}} \Eq \Exp{k}{n \Minus m}.\]
\end{proposition*}
\begin{proof}[forthel]
  We have $\Exp{k}{n} \Eq \Exp{k}{(n \Minus m) \Plus m} \Eq \Exp{k}{n \Minus m} \Mul \Exp{k}{m}$.
  Hence
  \[  \Frac{\Exp{k}{n}}{\Exp{k}{m}}
      \Eq \Frac{\Exp{k}{n \Minus m} \Mul \Exp{k}{m}}{\Exp{k}{m}}
      \Eq \Frac{\Exp{k}{n \Minus m} \Mul \Exp{k}{m}}{\One \Mul \Exp{k}{m}}
      \Eq \Frac{\Exp{k}{n \Minus m}}{\One}
      \Eq \Exp{k}{n \Minus m}. \]
\end{proof}
