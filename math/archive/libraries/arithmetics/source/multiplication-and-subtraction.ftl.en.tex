\begin{proposition*}[forthel]
  Let $n, m, k$ be natural numbers such that $n \Geq m$.
  Then $(n \Minus m) \Mul k \Eq (n \Mul k) \Minus (m \Mul k)$.
\end{proposition*}
\begin{proof}[forthel]
  We have
  \[  ((n \Minus m) \Mul k) \Plus (m \Mul k)                 \]
  \[    \Eq ((n \Minus m) \Plus m) \Mul k                       \]
  \[    \Eq n \Mul k                                   \]
  \[    \Eq ((n \Mul k) \Minus (m \Mul k)) \Plus (m \Mul k).  \]

  Hence $(n \Minus m) \Mul k \Eq (n \Mul k) \Minus (m \Mul k)$.
\end{proof}

\begin{proposition*}[forthel]
  Let $n, m, k$ be natural numbers such that $n \Geq m$.
  Then $k \Mul (n \Minus m) \Eq (k \Mul n) \Minus (k \Mul m)$.
\end{proposition*}
