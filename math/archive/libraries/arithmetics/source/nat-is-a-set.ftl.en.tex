\importmodule[libraries/set-theory]{axioms?infinity.ftl}

\begin{proposition*}[forthel]
  $\Naturals$ is a set.
\end{proposition*}
\begin{proof}[forthel]
  Define $f(n) \DefEq n \Plus \One$ for $n \Elem \Naturals$.
  Then $f$ is a map from $\Naturals$ to $\Naturals$.
  Hence we can take a subset $X$ of $\Naturals$ that is inductive regarding
  $\Zero$ and $f$.
  Then $\Zero \Elem X$ and for all $n \Elem X$ we have $n \Plus \One \Elem X$.
  Hence $X$ contains every natural number.
  Thus we have $\Naturals \Subclass X$ and $X \Subclass \Naturals$.
  Therefore $\Naturals \Eq X$.
  Consequently $\Naturals$ is a set.
\end{proof}
