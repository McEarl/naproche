\begin{smodule}{factorial.ftl}

\importmodule[libraries/arithmetics]{multiplication.ftl}

\symdef{Factorial}[args=1]{#1\comp!}

\begin{signature}[forthel,for=Factorial]
  Let $n$ be a natural number.
  $\emph{\Factorial{n}}$ is a natural number.
\end{signature}

\begin{axiom}[forthel,for=Factorial]
  $\Factorial{\Zero} \Eq \One$.
\end{axiom}

\begin{axiom}[forthel,for=Factorial]
  Let $n$ be a natural number.
  Then $\Factorial{n \Plus \One} \Eq \Factorial{n} \Mul (n \Plus \One)$.
\end{axiom}

\begin{proposition}[forthel]
  Let $n$ be a natural number.
  Then $\Factorial{n} \NotEq \Zero$.
\end{proposition}
\begin{proof}[forthel]
  Define $\Phi \DefEq \SClass{n'}{\Naturals}{\Factorial{n'} \NotEq \Zero}$.

  (1) $\Phi$ contains $\Zero$.
  Indeed $\Factorial{\Zero} \Eq \One \NotEq \Zero$.

  (2) For all $n' \Elem \Phi$ we have $n' \Plus \One \Elem \Phi$.
  \begin{proof}
    Let $n' \Elem \Phi$.
    We have $\Factorial{n' \Plus \One} \Eq (n' \Plus \One) \Mul \Factorial{n'}$.
    $n' \Plus \One$ and $\Factorial{n'}$ are nonzero.
    Hence $\Factorial{n' \Plus \One} \NotEq \Zero$.
  \end{proof}

  Thus $\Phi$ contains every natural number (by \sr{induction I}{induction}).
  Therefore $\Factorial{n} \NotEq \Zero$.
\end{proof}
