\begin{proposition}[forthel]
  Let $n$ be a natural number.
  Assume $n \Geq \Two$.
  Then $\Exp{n}{n} \Greater \Factorial{n}$.
\end{proposition}
\begin{proof}[forthel]
  (a) Define $\Phi \DefEq \SClass{n'}{\NatGeq{\Two}}{\Exp{n'}{n'} \Greater \Factorial{n'}}$.

  (1) $\Phi$ contains $\Two$.

  (2) For all $n' \Elem \Phi$ we have $n' \Plus \One \Elem \Phi$.
  \begin{proof}
    Let $n' \Elem \Phi$.
    Then $n' \Geq \Two$.

    Take $X \Eq \Exp{n' \Plus \One}{n'} \Mul (n' \Plus \One)$ and $Y \Eq \Exp{n'}{n'} \Mul (n' \Plus \One)$.
    Then $X \Greater Y$.
    \begin{proof}
      We have $n' \Plus \One \Greater n'$ and $n' \NotEq \Zero$.
      Thus $\Exp{n' \Plus \One}{n'} \Greater \Exp{n'}{n'}$.
      $n' \Plus \One$ is nonzero.
      Hence $\Exp{n' \Plus \One}{n'} \Mul (n' \Plus \One) \Greater \Exp{n'}{n'} \Mul (n' \Plus \One)$.
    \end{proof}

    Take $Z \Eq \Factorial{n'} \Mul (n' \Plus \One)$.
    Then $Y \Greater Z$.
    Indeed $\Exp{n'}{n'} \Greater \Factorial{n'}$ and $n' \Plus \One \NotEq \Zero$.

    Hence $\Exp{n' \Plus \One}{n' \Plus \One} \Eq X \Greater Y \Greater Z \Eq \Factorial{n' \Plus \One}$.
    Thus $\Exp{n' \Plus \One}{n' \Plus \One} \Greater \Factorial{n' \Plus \One}$.
  \end{proof}

  Therefore $\Phi$ contains every element of $\NatGeq{\Two}$ (by \sr{induction III}{induction}).
  Consequently $\Exp{n}{n} \Greater \Factorial{n}$ (by a).
\end{proof}

\begin{proposition}[forthel]
  Let $n$ be a natural number.
  Assume $n \Geq \Four$.
  Then $\Factorial{n} \Greater \Exp{\Two}{n}$.
\end{proposition}
\begin{proof}[forthel]
  (a) Define $\Phi \DefEq \SClass{n'}{\NatGeq{\Four}}{\Factorial{n'} \Greater \Exp{\Two}{n'}}$.

  (1) $\Phi$ contains $\Four$.
  \begin{proof}
    \[  \Factorial{\Four}                                  \]
    \[    \Eq \Four \Mul (\Three \Mul \Two)             \]
    \[    \Eq \Two \Mul (\Two \Mul (\Three \Mul \Two))   \]
    \[    \Eq \Three \Mul (\Two \Mul (\Two \Mul \Two))   \]
    \[    \Greater \Two \Mul (\Two \Mul (\Two \Mul \Two))   \]
    \[    \Eq \Exp{\Two}{\Four}.                          \]
  \end{proof}

  (2) For all $n' \Elem \Phi$ we have $n' \Plus \One \Elem \Phi$.
  \begin{proof}
    Let $n' \Elem \Phi$.
    Then $n' \Geq \Four$ and $\Factorial{n'} \Greater \Exp{\Two}{n'}$.

    Take $X \Eq \Factorial{n'} \Mul (n' \Plus \One)$ and $Y \Eq \Exp{\Two}{n'} \Mul (n' \Plus \One)$.
    Then $X \Greater Y$.

    Take $Z \Eq \Exp{\Two}{n'} \Mul \Two$.
    Then $Y \Greater Z$ (by \sn{preservation of ordering under left-multiplication}).
    Indeed $n' \Plus \One \Greater \Two$ and $\Exp{\Two}{n'} \NotEq \Zero$.

    Hence $\Factorial{n' \Plus \One} \Eq X \Greater Y \Greater Z \Eq \Exp{\Two}{n' \Plus \One}$.
    Thus $\Factorial{n' \Plus \One} \Greater \Exp{\Two}{n' \Plus \One}$.
  \end{proof}

  Therefore $\Phi$ contains every element of $\NatGeq{\Four}$ (by \sr{induction III}{induction}).
  Consequently $\Factorial{n} \Greater \Exp{\Two}{n}$ (by a).
\end{proof}
