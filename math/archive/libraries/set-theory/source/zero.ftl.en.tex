\begin{smodule}{zero.ftl}
\importmodule[libraries/set-theory]{ordinals.ftl}
\symdef{Zero}{0}

\begin{convention}[forthel,for=Zero]
  Let $\emph{\Zero}$ stand for $\EmptyClass$.
  Let $\alpha$ is \emph{nonzero} stand for $\alpha \NotEq \Zero$.
\end{convention}

\begin{proposition}[forthel]
  $\Zero$ is an ordinal.
\end{proposition}
\end{smodule}
