\begin{smodule}{successor-cardinals.ftl}
\importmodule[libraries/set-theory]{cardinals.ftl}
\importmodule[articles]{cantor.ftl}
\symdef{SuccCard}[args=1]{#1^{\comp+}}
\symdecl*{successor cardinal}

\begin{lemma}[forthel]
  For every ordinal $\alpha$ there exists a cardinal greater than $\alpha$.
\end{lemma}
\begin{proof}[forthel]
  Let $\alpha$ be an ordinal.
  Take $\kappa \Eq \Card{\Pow(\alpha)}$.
  Then $\kappa \Greater \Card{\alpha}$.

  Let us show that $\kappa \Greater \alpha$.
    Assume the contrary.
    Then $\Card{\Pow(\alpha)}
      \Eq \kappa
      \Leq \alpha$.
    Hence $\kappa
      \Eq \Card{\Pow(\alpha)}
      \Eq \Card{\Card{\Pow(\alpha)}}
      \Leq \Card{\alpha}$.
    Contradiction.
  End.
\end{proof}

\begin{definition}[forthel,for=SuccCard]
  Let $\kappa$ be a cardinal.
  $\emph{\SuccCard{\kappa}}$ is the cardinal such that $\kappa \Less \SuccCard{\kappa}$ and there is no cardinal $\nu$ such that $\kappa \Less \nu \Less \SuccCard{\kappa}$.
\end{definition}

\begin{definition}[forthel,for=successor cardinal]
  A \emph{successor cardinal} is a cardinal number $\kappa$ such that $\kappa \Eq \SuccCard{\nu}$ for some cardinal number $\nu$.
\end{definition}

\begin{proposition}[forthel]
  Let $\kappa$ be a cardinal.
  Then $\Card{\alpha} \Leq \kappa$ for every $\alpha \Elem \SuccCard{\kappa}$.
\end{proposition}
