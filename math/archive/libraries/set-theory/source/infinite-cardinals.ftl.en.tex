\symdef{Cardinals}{\mathbb C}

\begin{definition}[forthel,for=Cardinals]
  $\emph{\Cardinals}$ is the collection of all infinite cardinals.
\end{definition}

\begin{proposition}[forthel]
  Let $\alpha$ be an infinite ordinal.
  Then $\Card{\Succ(\alpha)} \Eq \Card{\alpha}$.
\end{proposition}
\begin{proof}[forthel]
  For any $\beta \Elem \Succ(\alpha)$ we have
  $\beta \Less \Infinity$ or $\Infinity \Leq \beta \Less \alpha$ or $\beta \Eq \alpha$.
  Define \[ f(\beta) \DefEq
    \begin{cases}
      \Succ(\beta) & : \beta \Less \Infinity
      \\
      \beta           & : \Infinity \Leq \beta \Less \alpha
      \\
      \Zero        & : \beta \Eq \alpha
    \end{cases} \]
  for $\beta \Elem \Succ(\alpha)$.

  Then $f$ is a map from $\Succ(\alpha)$ to $\alpha$.
  Indeed we can show that $f(\beta) \Elem \alpha$ for all
  $\beta \Elem \Succ(\alpha)$.
  \begin{proof}
    Let $\beta \Elem \Succ(\alpha)$.

    \begin{case}{$\beta \Less \Infinity$.}
      Then $f(\beta)
        \Eq \Succ(\beta)
        \Less \Infinity
        \Leq \alpha$.
    \end{case}

    \begin{case}{$\Infinity \Leq \beta \Less \alpha$.}
      Then $f(\beta)
        \Eq \beta
        \Less \alpha$.
    \end{case}

    \begin{case}{$\beta \Eq \alpha$.}
      Then $f(\beta)
        \Eq \Zero
        \Less \alpha$.
    \end{case}
  \end{proof}

  $f$ is surjective onto $\alpha$.
  Indeed we can show that for any $\beta \Elem \alpha$ there exists a
  $\gamma \Elem \Succ(\alpha)$ such that $\beta \Eq f(\gamma)$.
  \begin{proof}
    Let $\beta \Elem \alpha$.
    Then $\beta \Eq \Zero$ or $\Zero \Less \beta \Less \Infinity$ or $\beta \Geq \Infinity$.

    \begin{case}{$\beta \Eq \Zero$.}
      Then $\beta \Eq f(\alpha)$.
    \end{case}

    \begin{case}{$\Zero \Less \beta \Less \Infinity$.}
      Take an ordinal $\beta'$ such that $\beta \Eq \Succ(\beta')$.
      Then $\beta' \Less \Infinity$.
      Hence $\beta \Eq f(\beta')$.
    \end{case}

    \begin{case}{$\beta \Geq \Infinity$.}
      Then $\beta \Eq f(\beta)$.
    \end{case}
  \end{proof}

  $f$ is injective.
  Indeed we can show that for all $\beta, \gamma \Elem \Succ(\alpha)$ if
  $\beta \NotEq \gamma$ then $f(\beta) \NotEq f(\gamma)$.
  \begin{proof}
    Let $\beta, \gamma \Elem \Succ(\alpha)$.
    Assume $\beta \NotEq \gamma$.

    \begin{case}{$\beta \Less \Infinity$.}
      If $\gamma \Eq \alpha$ then
      $f(\beta)
        \Eq \Succ(\beta)
        \NotEq \Zero
        \Eq f(\gamma)$.
      If $\Infinity \Leq \gamma \Less \alpha$ then
      $f(\beta)
        \Eq \Succ(\beta)
        \Less \Infinity
        \Leq \gamma
        \Eq f(\gamma)$.
    \end{case}

    \begin{case}{$\Infinity \Leq \beta \Less \alpha$.}
      If $\gamma \Eq \alpha$ then
      $f(\beta)
        \Eq \beta
        \Geq \Infinity
        \Greater \Zero
        \Eq f(\gamma)$.
      If $\gamma \Less \Infinity$ then
      $f(\beta)
        \Eq \beta
        \Geq \Infinity
        \Greater \Succ(\gamma)
        \Eq f(\gamma)$.
    \end{case}

    \begin{case}{$\beta \Eq \alpha$.}
      If $\gamma \Less \Infinity$ then
      $f(\beta)
        \Eq \Zero
        \NotEq \Succ(\gamma)
        \Eq f(\gamma)$.
      If $\Infinity \Leq \gamma \Less \alpha$ then
      $f(\beta)
        \Eq \Zero
        \Less \Infinity
        \Leq \gamma
        \Eq f(\gamma)$.
    \end{case}
  \end{proof}

  Hence $f$ is a bijection between $\Succ(\alpha)$ and $\alpha$.
  Therefore $\Succ(\alpha)$ and $\alpha$ are equinumerous.
  Consequently $\Card{\Succ(\alpha)} \Eq \Card{\alpha}$.
\end{proof}

\begin{proposition}[forthel]
  Every infinite cardinal is a limit ordinal.
\end{proposition}
\begin{proof}[forthel]
  Let $\kappa$ be an infinite cardinal.
  Suppose that $\kappa$ is not a limit ordinal.
  $\kappa \NotEq \Zero$.
  Hence $\kappa$ is a successor ordinal.
  Indeed $\kappa$ is an ordinal.
  Thus we can take an ordinal $\alpha$ such that $\kappa \Eq \Succ(\alpha)$.
  We have $\alpha \Greater \kappa \Geq \Infinity$.
  Hence $\Card{\Succ(\alpha)} \Eq \Card{\alpha}$.
  Thus $\alpha \Less \Card{\kappa}$ and $\kappa$ is equinumerous to $\kappa$.
  Contradiction.
\end{proof}
