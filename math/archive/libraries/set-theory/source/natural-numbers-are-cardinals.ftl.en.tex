\begin{theorem}[forthel]
  Let $n \Elem \Infinity$.
  Then $\Card{n} \Eq n$.
\end{theorem}
\begin{proof}[forthel]
  Define $\Phi \DefEq \SClass{n'}{\Infinity}{\Card{n'} \Eq n'}$.

  (1) $\Zero \Elem \Phi$.
  Indeed $\Card{\Zero} \Eq \Card{\EmptyClass} \Eq \Zero$.

  (2) For all $n' \Elem \Phi$ we have $\Succ(n') \Elem \Phi$.
  \begin{proof}
    Let $n' \Elem \Phi$.
    Then $\Card{n'} \Eq n'$.
    We have $\Card{\Succ(n')} \Leq \Succ(n')$.

    Let us show that $\Succ(n') \Leq \Card{\Succ(n')}$.
      Assume the contrary.
      Then $\Card{\Succ(n')} \Less \Succ(n')$.
      Take a bijection $f$ between $\Card{\Succ(n')}$ and $\Succ(n')$.
      $\Card{\Succ(n')}$ is nonzero.
      Hence we can take a $m \Elem \Infinity$ such that $\Card{\Succ(n')} \Eq \Succ(m)$.
      Indeed $\Card{\Succ(n')} \Elem \Succ(\Succ(n')) \Elem \Infinity$.
      Then $\Inv{f}(n') \Leq m$.

      We can show that $\Inv{f}(n') \Less m$.
        Assume the contrary.
        Then $\Inv{f}(n') \Eq m$.
        $f \Rest m$ is a bijection between $m$ and $\Im{f}{m}$ (by \sn{bijectivity of restriction of injection}).
        Indeed $f$ is an injective map from $\Card{\Succ(n')}$ to $\Succ(n')$ and
        $m \Subclass \Card{\Succ(n')}$.
        We have $\Im{f}{m} \Subclass n'$ and $n' \Subclass \Im{f}{m}$.
        Hence $\Im{f}{m} \Eq n'$.
        Thus $f \Rest m$ is a bijection between $m$ and $n'$.
        Therefore $n'
          \Eq \Card{n'}
          \Leq m
          \Less \Card{\Succ(n')}
          \Less \Succ(n')$.
        Consequently $m \Eq n'$.
        Then we have $\Succ(n') \Eq \Card{\Succ(n')} \Less \Succ(n')$.
        Contradiction.
      End.

      Define \[ g(i) \DefEq
        \begin{cases}
          f(i)  & : i \NotEq \Inv{f}(n')
          \\
          f(m)  & : i \Eq \Inv{f}(n')
        \end{cases} \]
      for $i \Elem m$.

      $g$ is a map from $m$ to $n'$.
      Indeed we can show that $g(i) \Elem n'$ for each $i \Elem m$.
      \begin{proof}
        Let $i \Elem m$.
        We have $g(i) \Elem \Succ(n')$.
        Indeed $f(i), f(m) \Elem \Succ(n')$ and ($g(i) \Eq f(i)$ or $g(i) \Eq f(m)$).
        Hence if $g(i) \NotEq n'$ then $g(i) \Elem n'$.

        \begin{case}{$i \NotEq \Inv{f}(n')$.}
          Then $g(i) \Eq f(i) \Elem \Succ(n')$.
          If $g(i) \Eq n'$ then $f(i) \Eq n' \Eq f(\Inv{f}(n'))$.
          Hence if $g(i) \Eq n'$ then $i \Eq \Inv{f}(n')$.
          Thus $g(i) \NotEq n'$.
          Therefore $g(i) \Elem n'$.
        \end{case}

        \begin{case}{$i \Eq \Inv{f}(n')$.}
          Then $g(i)
            \Eq f(m)
            \NotEq f(\Inv{f}(n'))
            \Eq n'$.
          Hence $g(i) \Elem n'$.
        \end{case}
      \end{proof}

      $g$ is surjective onto $n'$.
      Indeed we can show that for all $k \Elem n'$ there exists a $l \Elem m$ such that $k \Eq g(l)$.
      \begin{proof}
        Let $k \Elem n'$.
        Then $\Inv{f}(k) \NotEq \Inv{f}(n')$.

        \begin{case}{$\Inv{f}(k) \Eq m$.}
          Then $k
            \Eq f(\Inv{f}(k))
            \Eq f(m)
            \Eq g(\Inv{f}(n'))$.
        \end{case}

        \begin{case}{$\Inv{f}(k) \NotEq m$.}
          Then $\Inv{f}(k) \Elem m$.
          Indeed $\Inv{f}(k) \Elem \Card{\Succ(n')} \Eq \Succ(m) \Eq m \Union \Singleton{m}$.
          Hence $k
            \Eq f(\Inv{f}(k))
            \Eq g(\Inv{f}(k))$.
        \end{case}
      \end{proof}

      $g$ is injective.
      Indeed we can show that for all $i, j \Elem m$ if $i \NotEq j$ then $g(i) \NotEq g(j)$.
      \begin{proof}
        Let $i, j \Elem m$.
        Assume $i \NotEq j$.

        \begin{case}{$i, j \NotEq \Inv{f}(n')$.}
          Then $g(i)
            \Eq f(i)
            \NotEq f(j)
            \Eq g(j)$.
        \end{case}

        \begin{case}{$i \Eq \Inv{f}(n')$.}
          Then $j \NotEq \Inv{f}(n')$.
          Hence $g(i)
            \Eq g(\Inv{f}(n'))
            \Eq f(m)
            \NotEq f(j)
            \Eq g(j)$.
          Indeed $m \NotEq j$.
        \end{case}

        \begin{case}{$j \Eq \Inv{f}(n')$.}
          Then $i \NotEq \Inv{f}(n')$.
          Hence $g(i)
            \Eq f(i)
            \NotEq f(m)
            \Eq g(\Inv{f}(n'))
            \Eq g(j)$.
          Indeed $i \NotEq m$.
        \end{case}
      \end{proof}
    End.
  \end{proof}

  Thus $\Infinity \Subclass \Phi$ (by \sn{transfinite induction III}).
  Consequently $n \Elem \Phi$.
  Therefore $\Card{n} \Eq n$.
\end{proof}

\begin{corollary}[forthel]
  Every natural number is a cardinal.
\end{corollary}
