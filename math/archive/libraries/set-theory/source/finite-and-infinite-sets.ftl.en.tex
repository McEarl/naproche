\begin{smodule}{finite-and-infinite-sets.ftl}

\importmodule[libraries/set-theory]{omega-is-cardinal.ftl}

\symdecl*{finite}
\symdecl*{infinite}

\begin{definition}[forthel,for=finite]
  Let $x$ be a set.
  $x$ is \emph{finite} iff $\Card{x} \Less \Infinity$.
\end{definition}

\begin{definition}[forthel,for=infinite]
  Let $x$ be a set.
  $x$ is \emph{infinite} iff $x$ is not finite.
\end{definition}

\begin{proposition}[forthel]
  Let $x$ be a set.
  Then $x$ is finite iff $\Card{x} \Eq n$ for some $n \Elem \Infinity$.
\end{proposition}

\begin{proposition}[forthel]
  Let $x$ be a set.
  Then $x$ is infinite iff $\Card{x} \Geq \Infinity$.
\end{proposition}
\begin{proof}[forthel]
  $\Card{x} \Geq \Infinity$ iff $\Card{x} \NotLess \Infinity$.
\end{proof}

\begin{proposition}[forthel]
  $\Infinity$ is infinite.
\end{proposition}
