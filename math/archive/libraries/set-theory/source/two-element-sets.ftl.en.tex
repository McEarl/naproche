\begin{smodule}{two-element-sets.ftl}

  \importmodule[libraries/set-theory]{two.ftl}
\importmodule[libraries/set-theory]{cardinals.ftl}

\begin{proposition}[forthel]
  Let $a, b$ be distinct objects.
  Then $\Card{\Pair{a}{b}} \Eq \Two$.
\end{proposition}
\begin{proof}[forthel]
  Define \[ f(x) \DefEq
    \begin{cases}
      \Zero  & x \Eq a
      \\
      \One   & x \Eq b
    \end{cases} \]
  for $x \Elem \Pair{a}{b}$.
  Then $f$ is a map from $\Pair{a}{b}$ to $\Two$.
  $f$ is injective and surjective onto $\Two$.
  Hence $f$ is a bijection between $\Pair{a}{b}$ and $\Two$.
  Consequently $\Pair{a}{b}$ and $\Two$ are equinumerous.
  Thus $\Card{\Pair{a}{b}} \Eq \Card{\Two} \Eq \Two$.
\end{proof}
