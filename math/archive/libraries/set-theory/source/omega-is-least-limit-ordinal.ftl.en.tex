\begin{proposition*}[forthel]
  Let $\lambda$ be a limit ordinal.
  Then \[ \Infinity \Leq \lambda. \]
\end{proposition*}
\begin{proof}[forthel]
  Assume the contrary.
  Then $\lambda \Less \Infinity$.
  Consequently $\lambda \Elem \Infinity$.
  Hence $\lambda \Eq \Zero$ or $\lambda \Eq \Succ(n)$ for some $n \Elem \Infinity$.
  Thus $\lambda$ is not a limit ordinal.
  Contradiction.
\end{proof}
