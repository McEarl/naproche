\begin{proposition}[forthel]
  Let $a$ be an object.
  Then $\Card{\Singleton{a}} \Eq \One$.
\end{proposition}
\begin{proof}[forthel]
  Define $f(x) \DefEq \Zero$ for $x \Elem \Singleton{a}$.
  Then $f$ is a map from $\Singleton{a}$ to $\One$.
  $f$ is injective and surjective onto $\One$.
  Hence $f$ is a bijection between $\Singleton{a}$ and $\One$.
  Consequently $\Singleton{a}$ and $\One$ are equinumerous.
  Thus $\Card{\Singleton{a}} \Eq \Card{\One} \Eq \One$.
\end{proof}
