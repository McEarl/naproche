\begin{smodule}{omega-is-cardinal.ftl}

\importmodule[libraries/set-theory]{natural-numbers-are-cardinals.ftl}

\begin{proposition*}[forthel]
  $\Card{\Infinity} \Eq \Infinity$.
\end{proposition*}
\begin{proof}[forthel]
  We have $\Card{\Infinity} \Leq \Infinity$.

  Let us show that $\Card{\Infinity}$ is not less than $\Infinity$.
    Assume the contrary.
    Then $\Card{\Infinity} \Elem \Infinity$.
    Take $n \Eq \Card{\Infinity}$ and a bijection $f$ between $n$ and $\Infinity$.

    Define \[ g(k) \DefEq
      \begin{cases}
        \Succ(f(k)) & : k \Less n
        \\
        \Zero           & : k \Eq n
      \end{cases} \]
    for $k \Elem \Succ(n)$.
    Then $g$ is a map from $\Succ(n)$ to $\Infinity$.
    Indeed we can show that $g(k) \Elem \Infinity$ for all $k \Elem \Succ(n)$.
      Let $k \Elem \Succ(n)$.
      \begin{case}{$k \Less n$.} \end{case}
      \begin{case}{$k \Eq n$.} \end{case}
    End.

    $g$ is injective.
    Indeed we can show that for all $k, k' \Elem \Succ(n)$ if $k \NotEq k'$
    then $g(k) \NotEq g(k')$. 
    \begin{proof}
      Let $k, k' \Elem \Succ(n)$.
      Assume $k \NotEq k'$.

      \begin{case}{$k, k' \Less n$.}
        Then $f(k) \NotEq f(k')$.
        Hence $\Succ(f(k)) \NotEq \Succ(f(k'))$.
        Thus $g(k) \NotEq g(k')$.
      \end{case}

      \begin{case}{$k \Less n$ and $k' \Eq n$.}
        We have $\Succ(f(k)) \NotEq \Zero$.
        Hence $g(k) \NotEq g(k')$.
      \end{case}

      \begin{case}{$k \Eq n$ and $k' \Less n$.}
        We have $\Succ(f(k')) \NotEq \Zero$.
        Hence $g(k) \NotEq g(k')$.
      \end{case}
    \end{proof}

    $g$ is surjective onto $\Infinity$.
    Indeed we can show that for any $m \Elem \Infinity$ there exists a $k \Elem \Succ(n)$ such that $m \Eq g(k)$.
    \begin{proof}
      Let $m \Elem \Infinity$.
      Then $\Inv{f}(m) \Elem n$.

      \begin{case}{$m \Eq \Zero$.}
        Then $m \Eq g(n)$.
      \end{case}

      \begin{case}{$m \NotEq \Zero$.}
        Take $m' \Elem \Infinity$ such that $m \Eq \Succ(m')$.
        Then $m
          \Eq \Succ(m')
          \Eq \Succ(f(\Inv{f}(m')))
          \Eq g(\Inv{f}(m'))$.
        Indeed $f(\Inv{f}(m')) \Eq m'$ and $\Inv{f}(m') \Less n$.
      \end{case}
    \end{proof}

    Hence $g$ is a bijection between $\Succ(n)$ and $\Infinity$.
    Then we have $n
      \Eq \Card{n}
      \Eq \Card{\Succ(n)}
      \Eq \Succ(n)$.
    Contradiction.
  End.
\end{proof}

\begin{proposition*}[forthel]
  $\Infinity$ is a cardinal.
\end{proposition*}
