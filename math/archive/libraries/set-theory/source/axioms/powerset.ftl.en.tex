\symdecl*{Powerset Axiom}

\begin{axiom}[forthel,title=Powerset,name=Powerset Axiom]
  Let $x$ be a set.
  Then $\Pow(x)$ is a set.
  Let the powerset of $x$ stand for $\Pow(x)$.
\end{axiom}
