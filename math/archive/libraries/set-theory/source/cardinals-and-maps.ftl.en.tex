\begin{proposition*}[forthel,name=cardinality of image under injection]
  Let $x, y$ be sets and $f \FromInto{x}{y}$ and $a \Subclass x$.
  Then $\Card{\Im{f}{a}} \Eq \Card{a}$.
\end{proposition*}
\begin{proof}[forthel]
  $f \Rest a$ is a bijection between $a$ and $\Im{f}{a}$.
  $\Im{f}{a}$ is a set.
  Hence $\Card{a} \Eq \Card{\Im{f}{a}}$.
\end{proof}

\begin{proposition*}[forthel]
  Let $\kappa$ be a cardinal and $x \Subclass \kappa$.
  Then $\Card{x} \Leq \kappa$.
\end{proposition*}
\begin{proof}[forthel]
  Assume $\Card{x} \Greater \kappa$.
  Then $\kappa \Subclass \Card{x}$.
  Take a bijection $f$ between $\Card{x}$ and $x$.
  Then $f \Rest \kappa$ is an injective map from $\kappa$ to $x$.
  $\Id{x}$ is an injective map from $x$ to $\kappa$.
  Hence $x$ and $\kappa$ are equinumerous (by \sr{Cantor-Schroeder-Bernstein}{Cantor-Schöder-Bernstein}).
  Indeed $x$ is a set.
  Thus $\Card{x} \Eq \kappa$.
  Contradiction.
\end{proof}

\begin{proposition*}[forthel,name=existence condition for injections]
  Let $x, y$ be sets.
  Then there exists an injective map from $x$ to $y$ iff $\Card{x} \Leq \Card{y}$.
\end{proposition*}
\begin{proof}[forthel]
  \begin{case}{there exists an injective map from $x$ to $y$.}
    Consider an injective map $f$ from $x$ to $y$.
    Take a bijection $g$ from $\Card{x}$ to $x$ and a bijection $h$ from $y$ to $\Card{y}$.
    Then $g$ is an injective map from $\Card{x}$ to $x$ and $h$ is an injective map from $y$ to $\Card{y}$.
    Hence $h \Comp f$ is an injective map from $x$ to $\Card{y}$.
    Thus $(h \Comp f) \Comp g$ is an injective map from $\Card{x}$ to $\Card{y}$.
    Therefore $\Card{x}
      \Eq \Card{\Card{x}}
      \Eq \Card{\Im{(h \Comp f) \Comp g}{\Card{x}}}$
    (by \sn{cardinality of cardinal number}, \sn{cardinality of image under injection}).
    We have $\Im{(h \Comp f) \Comp g}{\Card{x}} \Subclass \Card{y}$.
    Hence $\Card{x} \Leq \Card{y}$.
  \end{case}

  \begin{case}{$\Card{x} \Leq \Card{y}$.}
    Take a bijection $g$ from $x$ to $\Card{x}$ and a bijection $h$ from $\Card{y}$ to $y$.
    We have $\Card{x} \Subclass \Card{y}$.
    Hence $g$ is an injective map from $x$ to $\Card{y}$.
    Take $f \Eq h \Comp g$.
    Then $f$ is an injective map from $x$ to $y$.
    Indeed $f$ is injective.
    Indeed $h$ is an injective map from $\Card{y}$ to $y$.
  \end{case}
\end{proof}

\begin{corollary*}[forthel]
  Let $x$ be a set and $y \Subclass x$.
  Then $\Card{y} \Leq \Card{x}$.
\end{corollary*}
\begin{proof}[forthel]
  Define $f(v) \DefEq v$ for $v \Elem y$.
  Then $f$ is an injective map from $y$ to $x$.
  Hence $\Card{y} \Leq \Card{x}$.
\end{proof}

\begin{corollary*}[forthel]
  Let $x, y$ be sets such that $\Card{y} \Less \Card{x}$.
  Then $x \Diff y$ is nonempty.
\end{corollary*}
\begin{proof}[forthel]
  Assume the contrary.
  Then $x \Subclass y$.
  Hence $\Card{x} \Leq \Card{y}$.
  Contradiction.
\end{proof}

\begin{proposition*}[forthel,name=existence condition for surjections]
  Let $x, y$ be nonempty sets.
  Then there exists a surjective map from $x$ onto $y$ iff $\Card{x} \Geq \Card{y}$.
\end{proposition*}
\begin{proof}[forthel]
  \begin{case}{there exists a surjective map from $x$ onto $y$.}
    Consider a surjective map $f$ from $x$ onto $y$.
    Define $g(v) \DefEq$ ``choose $u \Elem x$ such that $f(u) \Eq v$ in $u$'' for $v \Elem y$.
    Then $g$ is an injective map from $y$ to $x$.
    Indeed we can show that $g$ is injective.
      Let $v, v' \Elem y$.
      Assume $g(v) \Eq g(v')$.
      Take $u \Elem x$ such that $f(u) \Eq v$ and $g(v) \Eq u$.
      Take $u' \Elem x$ such that $f(u') \Eq v'$ and $g(v') \Eq u'$.
      Then $v
        \Eq f(u)
        \Eq f(g(v))
        \Eq f(g(v'))
        \Eq f(u')
        \Eq v'$.
    End.
    Hence $\Card{x} \Geq \Card{y}$.
  \end{case}

  \begin{case}{$\Card{x} \Geq \Card{y}$.}
    Then we can take an injective map $f$ from $y$ to $x$.
    Then $\Inv{f}$ is a bijection between $\Range(f)$ and $y$.
    Consider an element $z$ of $y$.
    Define \[ g(u) \DefEq
      \begin{cases}
        \Inv{f}(u) & : u \Elem \Range(f) \\
        z         & : u \NotElem \Range(f)
      \end{cases} \]
    for $u \Elem x$.
    Then $g$ is a surjective map from $x$ onto $y$.
    Indeed we can show that every element of $y$ is a value of $g$.
      Let $v \Elem y$.
      Then $f(v) \Elem \Range(f)$.
      Hence $g(f(v)) \Eq \Inv{f}(f(v)) \Eq v$.
    End.
  \end{case}
\end{proof}

\begin{proposition*}[forthel]
  Let $x, y$ be nonempty sets.
  $\Card{x} \Less \Card{y}$ iff there exists an injective map from $x$ to $y$ and there exists no surjective map from $x$ onto $y$.
\end{proposition*}
\begin{proof}[forthel]
  There exists an injective map from $x$ to $y$ and there exists no surjective map from $x$ onto $y$ iff $\Card{x} \Leq \Card{y}$ and $\Card{x} \NotGeq \Card{y}$ (by \sn{existence condition for injections}, \sn{existence condition for surjections}).
  $\Card{x} \Leq \Card{y}$ and $\Card{x} \NotGeq \Card{y}$ iff $\Card{x} \Leq \Card{y}$ and $\Card{x} \NotEq \Card{y}$.
  $\Card{x} \Leq \Card{y}$ and $\Card{x} \NotEq \Card{y}$ iff $\Card{x} \Less \Card{y}$.
\end{proof}

\begin{proposition*}[forthel,name=cardinality of image]
  Let $x, y$ be sets and $f \FromTo{x}{y}$ and $a \Subclass x$.
  Then $\Card{\Im{f}{a}} \Leq \Card{a}$.
\end{proposition*}
\begin{proof}[forthel]
  \begin{case}{$a$ is empty.} \end{case}

  \begin{case}{$a$ is nonempty.}
    $f \Rest a$ is a surjective map from $a$ onto $\Im{f}{a}$.
    $\Im{f}{a}$ is nonempty.
    Hence $\Card{\Im{f}{a}} \Leq \Card{a}$ (by \sn{existence condition for surjections}).
    Indeed $a$ and $\Im{f}{a}$ are sets.
  \end{case}
\end{proof}
