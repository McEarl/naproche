    \inlineforthel{Let $x$ and $y$ \emph{agree} stand for $x \Eq y$.}
    \inlineforthel{Let $x$ \emph{agrees with $y$} stand for $x \Eq y$.}
  \end{signature*}
\end{fakeforthel}

\begin{fakeforthel}

    $\emph{x \NotEq y}$ iff $\Not x \Eq y$.
    \inlineforthel{Let $x$ and $y$ are \emph{distinct} stand for $x \NotEq y$.}
  \end{definition*}
\end{fakeforthel}
