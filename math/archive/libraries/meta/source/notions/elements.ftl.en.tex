\symdef{Elem}{\;\comp\in\;}
\symdef{NotElem}{\;\comp{\not\in}\;}

\begin{fakeforthel}
  \begin{signature*}[for={Elem,NotElem}]
    Let $A$ be a class.
    A \emph{element of $A$} is an object.
    \inlineforthel{Let a \emph{member of $X$} stand for an element of $X$.}
    \inlineforthel{Let $x$ \emph{belongs to $X$} stand for $x$ is an element of $X$.}
    \inlineforthel{Let $X$ \emph{contains $x$} stand for $x$ is an element of $X$.}
    \inlineforthel{Let $x$ is \emph{contained in $X$} stand for $x$ is an element of $X$.}
    \inlineforthel{Let $x$ \emph{lies in $X$} stand for $x$ is an element of $X$.}
    \inlineforthel{Let $x$ is \emph{in $X$} stand for $x$ is an element of $X$.}
    Let $\emph{a \Elem A}$ stand for $a$ is an element of $A$.
    Let $\emph{a \NotElem A}$ stand for $\Not a \Elem A$.
  \end{signature*}
\end{fakeforthel}

\end{smodule}
