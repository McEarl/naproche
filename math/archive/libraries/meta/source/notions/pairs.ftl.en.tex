\begin{fakeforthel}
  \begin{signature*}
    Let $a,b$ be objects.
    $\emph{(a,b)}$ is an object.
    \inlineforthel{Let the \emph{ordered pair of $a$ and $b$} stand for $(a,b)$.}
  \end{signature*}
\end{fakeforthel}

\begin{forthel}
  \begin{axiom*}[title=Ordered Pair Extensionality]
    Let $a,a',b,b'$ be objects.
    If $(a, b) \Eq (a', b')$ then $a \Eq a'$ and $b \Eq b'$.
  \end{axiom*}
\end{forthel}
