\begin{smodule}{domain.ftl}

\importmodule[libraries/meta]{relations?equality.ftl}

\symdef{Dom}{\textrm{dom}}

\begin{convention*}
  Let the \emph{domain of $f$} stand for $\Dom(f)$.
  Let $f$ is \emph{defined on $X$} stand for $\Dom(f) \Eq X$.
\end{convention*}
