\begin{fakeforthel}
  \begin{signature*}[for=Dom]
    Let $f$ be a map.
    $\emph{\Dom(f)}$ is a class.
    \inlineforthel{Let the \emph{domain of $f$} stand for $\Dom(f)$.}
  \end{signature*}
\end{fakeforthel}

\begin{forthel}
  \begin{definition*}
    Let $f$ be a map.
    $f$ is \emph{defined on $X$} iff $\Dom(f) \Eq X$.
  \end{definition*}
\end{forthel}
