  \begin{signature*}[for=Or]
    Let $\varphi, \psi$ be formulas.

  \begin{axiom*}[for=Or]
    Let $\varphi, \psi$ be formulas.
    $\varphi \Or \psi$ is true iff $\varphi$ is true or $\psi$ is true.
  \end{axiom*}
\end{fakeforthel}
