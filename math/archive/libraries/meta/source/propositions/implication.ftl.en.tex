\begin{smodule}{implication.ftl}

\symdef{Implies}{\;\comp\Longrightarrow\;}

\begin{fakeforthel}
  \begin{signature*}[for=Implies]
    Let $\varphi, \psi$ be formulas.
    $\emph{\varphi \Implies \psi}$ is a formula.
  \end{signature*}
\end{fakeforthel}

\begin{fakeforthel}
  \begin{axiom*}[for=Implies]
    Let $\varphi, \psi$ be formulas.
    $\varphi \Implies \psi$ is true iff $\varphi$ is true implies that $\psi$ is true.
  \end{axiom*}
\end{fakeforthel}
