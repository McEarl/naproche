\begin{smodule}{systems-of-sets.ftl}

\importmodule[libraries/foundations]{sub-and-supersets.ftl}

\symdef{UnionOver}[name=union over]{\bigcup\,}
\symdef{IntersectOver}[name=intersection over]{\bigcap\,}

\symdecl*{system of sets}
\symdecl*{system of nonempty sets}
\symdecl*{system of subsets}

\begin{definition*}[forthel,for=system of sets]
  A \emph{system of sets} is a class $X$ such that every element of $X$ is a set.
\end{definition*}

\begin{definition*}[forthel,for=system of nonempty sets]
  A \emph{system of nonempty sets} is a class $X$ such that every element of $X$ is a nonempty set.
\end{definition*}

\begin{definition*}[forthel,for=system of subsets]
  Let $A$ be a class.
  A \emph{system of subsets of $A$} is a class $X$ such that every element of $X$ is a subset of $A$.
\end{definition*}

\begin{proposition*}[forthel]
  Let $A$ be a class.
  Then $\EmptyClass$ is a system of subsets of $A$.
\end{proposition*}

\begin{proposition*}[forthel]
  Let $A$ be a class.
  Then $\Pow(A)$ is a system of subsets of $A$.
\end{proposition*}

\begin{proposition*}[forthel]
  Let $X, Y$ be systems of sets.
  Then $X \Union Y$ is a system of sets.
\end{proposition*}

\begin{proposition*}[forthel]
  Let $X, Y$ be systems of sets.
  Then $X \Intersect Y$ is a system of sets.
\end{proposition*}

\begin{proposition*}[forthel]
  Let $X, Y$ be systems of sets.
  Then $X \Diff Y$ is a system of sets.
\end{proposition*}

\begin{definition*}[forthel,for=union over]
  Let $X$ be a system of sets.
  $\emph{\UnionOver X} \DefEq \CClass{a}{a \Elem x\text{ for some }x \Elem X}$.
  Let \emph{union over $X$} stand for $\UnionOver X$.
\end{definition*}

\begin{proposition*}[forthel]
  $\UnionOver \EmptyClass \Eq \EmptyClass$.
\end{proposition*}
\begin{proof}[forthel]
  $\UnionOver \EmptyClass \Eq \CClass{a}{a \Elem x\text{ for some }x \Elem \EmptyClass}$.
  $\EmptyClass$ has no elements.
  Hence there is no object $a$ such that $a \Elem x$ for some $x \Elem \EmptyClass$.
  Thus $\UnionOver \EmptyClass \Eq \EmptyClass$.
\end{proof}

\begin{proposition*}[forthel]
  Let $x, y$ be sets.
  Then $\UnionOver \Pair{x}{y} \Eq x \Union y$.
\end{proposition*}
\begin{proof}[forthel]
  Let us show that $\UnionOver \Pair{x}{y} \Subclass x \Union y$.
    Let $a \Elem \UnionOver \Pair{x}{y}$.
    Then $a$ is contained in some element of $\Pair{x}{y}$.
    Hence $a \Elem x$ or $a \Elem y$.
    Thus $a \Elem x \Union y$.
  End.

  Let us show that $x \Union y \Subclass \UnionOver \Pair{x}{y}$.
    Let $a \Elem x \Union y$.
    Then $a \Elem x$ or $a \Elem y$.
    Hence $a$ is contained in some element of $\Pair{x}{y}$.
    Therefore $a \Elem \UnionOver \Pair{x}{y}$.
  End.
\end{proof}

\begin{corollary*}[forthel]
  Let $x$ be a set.
  Then $\UnionOver \Singleton{x} \Eq x$.
\end{corollary*}

\begin{definition*}[forthel,for=intersection over]
  Let $X$ be a system of sets.
  $\emph{\IntersectOver X} \DefEq \CClass{a}{a \Elem x\text{ for all }x \Elem X}$.
  Let the \emph{intersection over $X$} stand for $\IntersectOver X$.
\end{definition*}

\begin{proposition*}[forthel]
  $\IntersectOver \EmptyClass$ is the class of all objects.
\end{proposition*}
\begin{proof}[forthel]
  Define $V \DefEq \CClass{x}{x\text{ is an object}}$.
  We have $\IntersectOver \EmptyClass \Subclass V$.
  Indeed every element of $\IntersectOver \EmptyClass$ is an object.

  Let us show that $V \Subclass \IntersectOver \EmptyClass$.
    Let $a \Elem V$.
    Then $a$ is an object.
    For every $x \Elem \EmptyClass$ we have $a \Elem x$.
    Indeed $\EmptyClass$ has no elements.
    Thus $a \Elem \IntersectOver \EmptyClass$.
  End.
\end{proof}

\begin{proposition*}[forthel]
  Let $x, y$ be sets.
  Then $\IntersectOver \Pair{x}{y} \Eq x \Intersect y$.
\end{proposition*}
\begin{proof}[forthel]
  Let us show that $\IntersectOver \Pair{x}{y} \Subclass x \Intersect y$.
    Let $a \Elem \IntersectOver \Pair{x}{y}$.
    Then $a$ is contained in every element of $\Pair{x}{y}$.
    Hence $a \Elem x$ and $a \Elem y$.
    Thus $a \Elem x \Intersect y$.
  End.

  Let us show that $x \Intersect y \Subclass \IntersectOver \Pair{x}{y}$.
    Let $a \Elem x \Intersect y$.
    Then $a \Elem x$ and $a \Elem y$.
    Hence $a$ is contained in every element of $\Pair{x}{y}$.
    Therefore $a \Elem \IntersectOver \Pair{x}{y}$.
  End.
\end{proof}

\begin{corollary*}[forthel]
  Let $x$ be a set.
  Then $\IntersectOver \Singleton{x} \Eq x$.
\end{corollary*}
