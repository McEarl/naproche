\begin{smodule}{segments-of-natural-numbers.ftl}

\importmodule[libraries/arithmetics]{ordering.ftl}

\symdef{NatSegment}[args=2]{\comp{\{}#1\comp{,\dots,}#2\comp{\}}}

\begin{definition}[forthel,for=NatSegment]
  Let $n, m$ be natural numbers.
  $\emph{\NatSegment{n}{m}} \DefEq \SClass{k}{\Naturals}{n \Leq k \Leq m}$.
\end{definition}

\begin{proposition}[forthel]
  Let $n, m$ be natural numbers.
  If $\NatSegment{\One}{n} \Eq \NatSegment{\One}{m}$ then $n \Eq m$.
\end{proposition}
\begin{proof}[forthel]
  Assume $\NatSegment{\One}{n} \Eq \NatSegment{\One}{m}$.

  \begin{case}{$n \Eq \Zero$.}
    Then $\NatSegment{\One}{n} \Eq \EmptyClass$.
    Thus $\NatSegment{\One}{m} \Eq \EmptyClass$.
    Hence there exists no $k \Elem \Naturals$ such that $\One \Leq k \Leq m$.
    Therefore $m \Eq \Zero$.
    Consequently $n \Eq m$.
  \end{case}

  \begin{case}{$m \Eq \Zero$.}
    Then $\NatSegment{\One}{m} \Eq \EmptyClass$.
    Thus $\NatSegment{\One}{n} \Eq \EmptyClass$.
    Hence there exists no $k \Elem \Naturals$ such that $\One \Leq k \Leq n$.
    Therefore $n \Eq \Zero$.
    Consequently $n \Eq m$.
  \end{case}

  \begin{case}{$n, m \Geq \One$.}
    For all $k \Elem \Naturals$ we have $\One \Leq k \Leq n$ iff $\One \Leq k \Leq m$.
    Hence for all $k \Elem \Naturals$ we have $k \Leq n$ iff $k \Leq m$.

    Let us show by contradiction that $n \Eq m$.
      Suppose $n \NotEq m$.
      Then $n \Greater m$ or $m \Greater n$.

      \begin{case}{$n \Greater m$.}
        Take $k \Eq m \Plus \One$.
        Then $k \Leq n$ and $k \NotLeq m$.
        Hence it is wrong that $k \Leq n$ iff $k \Leq m$.
        Contradiction.
      \end{case}

      \begin{case}{$m \Greater n$.}
        Take $k \Eq n \Plus \One$.
        Then $k \Leq m$ and $k \NotLeq m$.
        Hence it is wrong that $k \Leq n$ iff $k \Leq m$.
        Contradiction.
      \end{case}
    End.
  \end{case}
\end{proof}

\begin{proposition}[forthel]
  Let $n$ be a natural number.
  Then $\NatSegment{\One}{n \Plus \One} \Eq \NatSegment{\One}{n} \Union \Singleton{n \Plus \One}$.
\end{proposition}
\begin{proof}[forthel]
  We have
  $\NatSegment{\One}{n \Plus \One} \Subclass \NatSegment{\One}{n} \Union \Singleton{n \Plus \One}$ and $\NatSegment{\One}{n} \Union \Singleton{n \Plus \One} \Subclass \NatSegment{\One}{n \Plus \One}$.
\end{proof}
