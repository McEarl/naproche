\begin{proposition*}[forthel]
  Let $A, B, C$ be classes.
  IF $A \Subclass B$ then $A \Prod C \Subclass B \Prod C$.
\end{proposition*}
\begin{proof}[forthel]
  Assume $A \Subclass B$.
  Let $x \Elem A \Prod C$.
  Take $a \Elem A$ and $c \Elem C$ such that $x \Eq (a, c)$.
  Then $a \Elem B$.
  Hence $(a, c) \Elem B \Prod C$.
\end{proof}

\begin{proposition*}[forthel]
  Let $A, A', B, B'$ be classes.
  Assume that $A$ and $A'$ are nonempty.
  Then $(A \Prod A') \Subclass (B \Prod B')$ iff $A \Subclass B$ and $A' \Subclass B'$.
\end{proposition*}
\begin{proof}[forthel]
  \begin{case}{$(A \Prod A') \Subclass (B \Prod B')$.}
    Let us show that for all $a \Elem A$ and all $a' \Elem A'$ we have $a \Elem B$ and $a' \Elem B'$.
      Let $a \Elem A$ and $a' \Elem A'$.
      Then $(a, a') \Elem A \Prod A'$.
      Hence $(a, a') \Elem B \Prod B'$.
      Thus $a \Elem B$ and $a' \Elem B'$.
    End.
  \end{case}

  \begin{case}{$A \Subclass B$ and $A' \Subclass B'$.}
    Let $x \Elem A \Prod A'$.
    Take $a \Elem A$ and $a' \Elem A'$ such that $x \Eq (a, a')$.
    Then $a \Elem B$ and $a' \Elem B'$.
    Hence $(a, a') \Elem B \Prod B'$.
  \end{case}
\end{proof}

\begin{proposition*}[forthel]
  Let $A, B, C$ be classes.
  Then $(A \Union B) \Prod C \Eq (A \Prod C) \Union (B \Prod C)$.
\end{proposition*}
\begin{proof}[forthel]
  Let us show that $((A \Union B) \Prod C) \Subclass (A \Prod C) \Union (B \Prod C).$ %!
    Let $x \Elem (A \Union B) \Prod C$.
    Take $y \Elem A \Union B$ and $c \Elem C$ such that $x \Eq (y, c)$.
    Then $y \Elem A$ or $y \Elem B$.
    If $y \Elem A$ then $x \Elem A \Prod C$ and if $y \Elem B$ then $x \Elem B \Prod C$.
    Hence $x \Elem A \Prod C$ or $x \Elem B \Prod C$.
    Thus $x \Elem (A \Prod C) \Union (B \Prod C)$.
  End.

  Let us show that $((A \Prod C) \Union (B \Prod C)) \Subclass (A \Union B) \Prod C$. %!
    Let $x \Elem (A \Prod C) \Union (B \Prod C)$.
    Then $x \Elem A \Prod C$ or $x \Elem B \Prod C$.
    Take objects $y, c$ such that $x \Eq (y, c)$.
    Then ($y \Elem A$ or $y \Elem B$) and $c \Elem C$.
    Hence $y \Elem A \Union B$.
    Thus $x \Elem (A \Union B) \Prod C$.
  End.
\end{proof}

\begin{proposition*}[forthel]
  Let $A, B, C$ be classes.
  Then $A \Prod (B \Union C) \Eq (A \Prod B) \Union (A \Prod C)$.
\end{proposition*}
\begin{proof}[forthel]
  Let us show that $A \Prod (B \Union C) \Subclass (A \Prod B) \Union (A \Prod C)$.
    Let $x \Elem A \Prod (B \Union C)$.
    Take $a \Elem A$ and $y \Elem B \Union C$ such that $x \Eq (a, y)$.
    Then $y \Elem B$ or $y \Elem C$.
    Hence $x \Elem A \Prod B$ or $x \Elem A \Prod C$.
    Indeed if $y \Elem B$ then $x \Elem A \Prod B$ and if $y \Elem C$ then $x \Elem A \Prod C$.
    Thus $x \Elem (A \Prod B) \Union (A \Prod C)$.
  End.

  Let us show that $((A \Prod B) \Union (A \Prod C)) \Subclass A \Prod (B \Union C)$.
    Let $x \Elem (A \Prod B) \Union (A \Prod C)$.
    Then $x \Elem A \Prod B$ or $x \Elem A \Prod C$.
    Take objects $a, y$ such that $x \Eq (a, y)$.
    Then $a \Elem A$ and ($y \Elem B$ or $y \Elem C$).
    Hence $x \Elem A \Prod (B \Union C)$.
  End.
\end{proof}

\begin{proposition*}[forthel]

  Then $(A \Intersect B) \Prod C \Eq (A \Prod C) \Intersect (B \Prod C)$.
\end{proposition*}
\begin{proof}[forthel]

    Then $y \Elem A$ and $y \Elem B$.
    Hence $x \Elem A \Prod C$ and $x \Elem B \Prod C$.
    Thus $x \Elem (A \Prod C) \Intersect (B \Prod C)$.
  End.

  Let us show that $((A \Prod C) \Intersect (B \Prod C)) \Subclass (A \Intersect B) \Prod C$. %!

    Take objects $y, z$ such that $x \Eq (y, z)$.
    Then ($y \Elem A$ and $y \Elem B$) and $z \Elem C$.
    Hence $y \Elem A \Intersect B$.
    Thus $x \Elem (A \Intersect B) \Prod C$.
  End.
\end{proof}

\begin{proposition*}[forthel]
  Let $A, B, C$ be classes.
  Then $(A \Diff B) \Prod C \Eq (A \Prod C) \Diff (B \Prod C)$.
\end{proposition*}
\begin{proof}[forthel]

    Then $a \Elem A$ and $a \NotElem B$.
    Hence $x \Elem A \Prod C$ and $x \NotElem B \Prod C$.
    Thus $x \Elem (A \Prod C) \Diff (B \Prod C)$.
  End.

  Let us show that $((A \Prod C) \Diff (B \Prod C)) \Subclass (A \Diff B) \Prod C$. %!
    Let $x \Elem (A \Prod C) \Diff (B \Prod C)$.

    Take $a \Elem A$ and $c \Elem C$ such that $x \Eq (a, c)$.
    Then $a \NotElem B$.
    Indeed if $a \Elem B$ then $x \Elem B \Prod C$.
    Hence $a \Elem A \Diff B$.
    Thus $x \Elem (A \Diff B) \Prod C$.
  End.

\begin{proposition*}[forthel]
  Let $A, A', B, B'$ be classes.
  Assume that $A$ and $A'$ are nonempty or $B$ and $B'$ are nonempty.
  Then $(A \Prod A') \Eq (B \Prod B')$ iff $A \Eq B$ and $A' \Eq B'$.
\end{proposition*}
\begin{proof}[forthel]
  \begin{case}{$A \Prod A' \Eq B \Prod B'$.}
    Then $A$ and $A'$ are nonempty iff $B$ and $B'$ are nonempty.

    Let us show that for all $a \Elem A$ and all $a' \Elem A'$ we have $a \Elem B$ and $a' \Elem B'$.
      Let $a \Elem A$ and $a' \Elem A'$.
      Then $(a, a') \Elem A \Prod A'$.
      Hence we can take $x \Elem B \Prod B'$ such that $x \Eq (a, a')$.
      Thus $a \Elem B$ and $a' \Elem B'$.
    End.

    Therefore $A \Subclass B$ and $A' \Subclass B'$.
    Indeed $A$ and $A'$ are nonempty.

    Let us show that for all $b \Elem B$ and all $b' \Elem B'$ we have $b \Elem A$ and $b' \Elem A'$.
      Let $b \Elem B$ and $b' \Elem B'$.
      Then $(b, b') \Elem B \Prod B'$.
      Hence we can take $x \Elem A \Prod A'$ such that $x \Eq (b, b')$.
      Thus $(b, b') \Elem A \Prod A'$.
    End.

    Therefore $B \Subclass A$ and $B' \Subclass A'$.
    Indeed $B$ and $B'$ are nonempty.
  \end{case}

  \begin{case}{$A \Eq B$ and $A' \Eq B'$.} \end{case}
\end{proof}

\begin{proposition*}[forthel]
  Let $A, A', B, B'$ be classes.
  Then $(A \Prod B) \Intersect (A' \Prod B') \Eq (A \Intersect A') \Prod (B \Intersect B')$.
\end{proposition*}
\begin{proof}[forthel]
  Let us show that $((A \Prod B) \Intersect (A' \Prod B')) \Subclass (A \Intersect A') \Prod (B \Intersect B')$. %!
    Let $x \Elem (A \Prod B) \Intersect (A' \Prod B')$.
    Then $x \Elem A \Prod B$ and $x \Elem A' \Prod B'$.
    Take objects $a, b$ such that $x \Eq (a, b)$.
    Then $a \Elem A, A'$ and $b \Elem B, B'$.
    Hence $a \Elem A \Intersect A'$ and $b \Elem B \Intersect B'$.
    Thus $x \Elem (A \Intersect A') \Prod (B \Intersect B')$.
  End.

  Let us show that $(A \Intersect A') \Prod (B \Intersect B') \Subclass (A \Prod B) \Intersect (A' \Prod B')$.
    Let $x \Elem (A \Intersect A') \Prod (B \Intersect B')$.
    Take elements $a, b$ such that $x \Eq (a, b)$.
    Then $a \Elem A \Intersect A'$ and $b \Elem B \Intersect B'$.
    Hence $a \Elem A, A'$ and $b \Elem B, B'$.
    Thus $x \Elem A \Prod B$ and $x \Elem A' \Prod B'$.
    Therefore $x \Elem (A \Prod B) \Intersect (A' \Prod B')$.
  End.
\end{proof}

\begin{proposition*}[forthel]
  Let $A, A', B, B'$ be classes.
  Then $(A \Prod B) \Union (A' \Prod B') \Subclass (A \Union A') \Prod (B \Union B')$.
\end{proposition*}
\begin{proof}[forthel]
  Let $x \Elem (A \Prod B) \Union (A' \Prod B')$.
  Then $x \Elem A \Prod B$ or $x \Elem A' \Prod B'$.
  Take objects $a, b$ such that $x \Eq (a, b)$.
  Then ($a \Elem A$ or $a \Elem A'$) and ($b \Elem B$ or $b \Elem B'$).
  Hence $a \Elem A \Union A'$ and $b \Elem B \Union B'$.
  Thus $x \Elem (A \Union A') \Prod (B \Union B')$.
\end{proof}

\begin{proposition*}[forthel]
  Let $A, A', B, B'$ be classes.
  Then $(A \Prod B) \Diff (A' \Prod B') \Eq (A \Prod (B \Diff B')) \Union ((A \Diff A') \Prod B)$.
\end{proposition*}
\begin{proof}[forthel]
  Let us show that $((A \Prod B) \Diff (A' \Prod B')) \Subclass (A \Prod (B \Diff B')) \Union ((A \Diff A') \Prod B)$.
    Let $x \Elem (A \Prod B) \Diff (A' \Prod B')$.
    Then $x \Elem A \Prod B$ and $x \NotElem A' \Prod B'$.
    Take $a \Elem A$ and $b \Elem B$ such that $x \Eq (a, b)$.
    Then it is wrong that $a \Elem A'$ and $b \Elem B'$.
    Hence $a \NotElem A'$ or $b \NotElem B'$.
    Thus $a \Elem A \Diff A'$ or $b \Elem B \Diff B'$.
    Therefore $x \Elem A \Prod (B \Diff B')$ or $x \Elem (A \Diff A') \Prod B$.
    Hence we have $x \Elem (A \Prod (B \Diff B')) \Union ((A \Diff A') \Prod B)$.
  End.

  Let us show that $(A \Prod (B \Diff B')) \Union ((A \Diff A') \Prod B) \Subclass (A \Prod B) \Diff (A' \Prod B')$.
    Let $x \Elem (A \Prod (B \Diff B')) \Union ((A \Diff A') \Prod B)$.
    Then $x \Elem (A \Prod (B \Diff B'))$ or $x \Elem ((A \Diff A') \Prod B)$.
    Take elements $a,b$ such that $x \Eq (a, b)$.
    Indeed $x \Elem X \Prod Y$ for some classes $X,Y$.
    Then ($a \Elem A$ and $b \Elem B \Diff B'$) or ($a \Elem A \Diff A'$ and $b \Elem B$).

    \begin{case}{$a \Elem A$ and $b \Elem B \Diff B'$.}
      Then $a \Elem A$ and $b \Elem B$.
      Hence $x \Elem A \Prod B$.
      We have $b \NotElem B'$.
      Thus $x \NotElem A' \Prod B'$.
      Therefore $x \Elem (A \Prod B) \Diff (A' \Prod B')$.
    \end{case}

    \begin{case}{$a \Elem A \Diff A'$ and $b \Elem B$.}
      Then $a \Elem A$ and $b \Elem B$.
      Hence $x \Elem A \Prod B$.
      We have $a \NotElem A'$.
      Thus $x \NotElem A' \Prod B'$.
      Therefore $x \Elem (A \Prod B) \Diff (A' \Prod B')$.
    \end{case}
  End.
\end{proof}
