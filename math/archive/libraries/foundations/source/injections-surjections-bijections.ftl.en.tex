\symdef{FromOnto}[args=2]{\,\comp:\,#1\,\comp\twoheadrightarrow\,#2}
\symdef{FromInto}[args=2]{\,\comp:\,#1\,\comp\hookrightarrow\,#2}

\begin{definition*}[forthel,for=surjective]
  Let $f$ be a map and $B$ be a class.
  $f$ is \emph{surjective onto $B$} iff $\Range(f) \Eq B$.
  Let $f$ \emph{surjects onto $B$} stand for $f$ is surjective onto $B$.
  Let a \emph{surjective map onto $B$} stand for a map that is surjective onto $B$.
\end{definition*}

\begin{definition*}[forthel,for={surjective,FromOnto}]
  Let $A, B$ be classes.
  A \emph{surjective map from $A$ to $B$} is a map of $A$ that is surjective onto $B$.
  Let a \emph{surjective map from $A$ onto $B$} stand for a surjective map from $A$ to $B$.
  Let $\emph{f \FromOnto{A}{B}}$ stand for $f$ is a surjective map from $A$ onto $B$.
\end{definition*}

\begin{proposition*}[forthel]
  Let $B$ be a class and $f$ be a map to $B$.
  $f$ is surjective onto $B$ iff every element of $B$ is a value of $f$.
\end{proposition*}
\begin{proof}[forthel]
  \begin{case}{$f$ is surjective onto $B$.}
    Then $B \Eq \Range(f)$.
    Let $b$ be an element of $B$.
    Then $b \Elem \Range(f)$.
    Hence $b$ is a value of $f$.
  \end{case}

  \begin{case}{every element of $B$ is a value of $f$.}
    Let us show that $B \Subclass \Range(f)$.
      Let $b \Elem B$.
      Then $b$ is a value of $f$.
      Hence $b \Elem \Range(f)$.
    End.

    Let us show that $\Range(f) \Subclass B$.
      Let $b \Elem \Range(f)$.
      Then $b$ is a value of $f$.
      Hence $b \Elem B$.
    End.
  \end{case}
\end{proof}

\begin{definition*}[forthel,for={injective,FromInto}]
  Let $f$ be a map.
  $f$ is \emph{injective} iff for all $a, a' \Elem \Dom(f)$ if $f(a) \Eq f(a')$ then $a \Eq a'$.
  Let $\emph{f \FromInto{A}{B}}$ stand for $f$ is an injective map from $A$ to $B$.
\end{definition*}

\begin{definition*}[forthel,for=bijection]
  Let $A, B$ be classes.
  A \emph{bijection between $A$ and $B$} is an injective map from $A$ to $B$ that is surjective onto $B$.
  Let a \emph{bijection from $A$ to $B$} stand for a bijection between $A$ and $B$.
\end{definition*}

\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f \FromInto{A}{B}$.
  Then $f$ is a bijection between $A$ and $\Range(f)$.
\end{proposition*}
\begin{proof}[forthel]
  $f$ is injective and surjects onto $\Range(f)$.
  Hence $f$ is a bijection between $A$ and $\Range(f)$.
\end{proof}

\begin{definition*}[forthel,for=permutation]
  Let $A$ be a class.
  A \emph{permutation of $A$} is a bijection between $A$ and $A$.
\end{definition*}

\begin{proposition*}[forthel]
  Let $A$ be a class.
  Then $\Id{A}$ is a permutation of $A$.
\end{proposition*}
\begin{proof}[forthel]
  (1) $\Id{A}$ is a map on $A$.

  (2) $\Id{A}$ is surjective onto $A$.
  \begin{proof}
    Let $a \Elem A$.
    Then $a \Eq \Id{A}(a)$.
    Hence $a \Elem \Range(\Id{A})$.
  \end{proof}

  (3) $\Id{A}$ is injective.
  \begin{proof}
    Let $a, a' \Elem A$.
    Assume $\Id{A}(a) \Eq \Id{A}(a')$.
    Then $a \Eq a'$.
  \end{proof}
\end{proof}

\begin{proposition*}[forthel,name=surjectivity of composition of surjections]
  Let $A, B, C$ be classes and $f \FromOnto{A}{B}$ and $g \FromOnto{B}{C}$.
  Then $g \Comp f$ is a surjective map from $A$ onto $C$.
\end{proposition*}
\begin{proof}[forthel]
  $g \Comp f$ is a map of $A$.

  Let us show that $g \Comp f$ is surjective onto $C$.
    Let $c \Elem C$.
    Take $b \Elem B$ such that $c \Eq g(b)$.
    Take $a \Elem A$ such that $b \Eq f(a)$.
    Then $c \Eq g(f(a)) \Eq (g \Comp f)(a)$.
  End.
\end{proof}

\begin{proposition*}[forthel,name=injectivity of composition of injections]
  Let $A, B, C$ be classes and $f \FromInto{A}{B}$ and $g \FromInto{B}{C}$.
  Then $g \Comp f$ is an injective map from $A$ to $C$.
\end{proposition*}
\begin{proof}[forthel]
  $g \Comp f$ is a map of $A$.

  Let us show that $g \Comp f$ is injective.
    Let $a, a' \Elem A$.
    Assume $(g \Comp f)(a) \Eq (g \Comp f)(a')$.
    Then $g(f(a)) \Eq g(f(a'))$.
    Hence $f(a) \Eq f(a')$.
    Indeed $f(a), f(a') \Elem B$.
    Thus $a \Eq a'$.
  End.
\end{proof}

\begin{corollary*}[forthel,name=bijectivity of composition of bijections]
  Let $A, B, C$ be classes.
  Let $f$ be a bijection between $A$ and $B$ and $g$ be a bijection between $B$ and $C$.
  Then $g \Comp f$ is a bijection between $A$ and $C$.
\end{corollary*}
\begin{proof}[forthel]
  $g \Comp f$ is an injective map from $A$ to $C$ (by \sn{injectivity of composition of injections}).
  $g \Comp f$ is a surjective map from $A$ onto $C$ (by \sn{surjectivity of composition of surjections}).
\end{proof}

\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f \FromInto{A}{B}$ and $X \Subclass A$.
  Then $f \Rest X$ is injective.
\end{proposition*}
\begin{proof}[forthel]
  Let $a, a' \Elem X$.
  Assume $(f \Rest X)(a) \Eq (f \Rest X)(a')$.
  Then $f(a) \Eq f(a')$.
  Hence $a \Eq a'$.
\end{proof}

\begin{proposition*}[forthel,name=bijectivity of restriction of injection]
  Let $A, B$ be classes and $f \FromInto{A}{B}$ and $X \Subclass A$.
  Then $f \Rest X$ is a bijection between $X$ and $\Im{f}{X}$.
\end{proposition*}

\begin{corollary*}[forthel]
  Let $A, B$ be classes and $f \FromInto{A}{B}$.
  Then $f$ is a bijection between $A$ and $\Im{f}{A}$.
\end{corollary*}
