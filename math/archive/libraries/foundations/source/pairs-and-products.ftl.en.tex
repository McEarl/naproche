\importmodule[libraries/foundations]{classes.ftl}

\symdef{Prod}{\,\times\,}

\symdecl*{ordered pai}
\symdecl*{Cartesian product}
\symdecl*{direct product}

\begin{definition}[forthel,for=ordered pair]
  A \emph{pair} is an object $p$ such that $p \Eq (a, b)$ for some objects $a, b$.
  Let an \emph{ordered pair} stand for a pair.
\end{definition}

\begin{definition}[forthel,for={Prod,Cartesian product,direct product}]
  Let $A, B$ be classes.
  $\emph{A \Prod B} \DefEq \CClass{(a, b)}{a \Elem A\text{ and }b \Elem B}$.
  Let the \emph{Cartesian product of $A$ and $B$} stand for $A \Prod B$.
  Let the \emph{direct product of $A$ and $B$} stand for $A \Prod B$.
\end{definition}

\begin{proposition}[forthel]
  Let $A, B$ be classes and $a, b$ be objects.
  Then $(a, b) \Elem A \Prod B$ iff $a \Elem A$ and $b \Elem B$.
\end{proposition}
\begin{proof}[forthel]
  \begin{case}{$(a, b) \Elem A \Prod B$.}
    We can take $a' \Elem A$ and $b' \Elem B$ such that $(a, b) \Eq (a', b')$.
    Then $a \Eq a'$ and $b \Eq b'$.
    Hence $a \Elem A$ and $b \Elem B$.
  \end{case}

  \begin{case}{$a \Elem A$ and $b \Elem B$.}
    $a$ and $a$ are objects.
    Hence $(a, b)$ is an object.
    Therefore $(a, b) \Elem A \Prod B$.
  \end{case}
\end{proof}

\begin{proposition}[forthel]
  Let $A, B$ be classes.
  Then $A \Prod B$ is empty iff $A$ is empty or $B$ is empty.
\end{proposition}
\begin{proof}[forthel]
  \begin{case}{$A \Prod B$ is empty.}
    Assume that $A$ and $B$ are nonempty.
    Then we can take an element $a$ of $A$ and an element $b$ of $B$.
    Then $(a, b) \Elem A \Prod B$.
    Contradiction.
  \end{case}

  \begin{case}{$A$ is empty or $B$ is empty.}
    Assume that $A \Prod B$ is nonempty.
    Then we can take an element $c$ of $A \Prod B$.
    Then $c \Eq (a, b)$ for some $a \Elem A$ and some $b \Elem B$.
    Hence $A$ and $B$ are nonempty.
    Contradiction.
  \end{case}
\end{proof}

\begin{proposition}[forthel]
  Let $a, b$ be objects.
  Then $\Singleton{a} \Prod \Singleton{b} \Eq \Singleton{(a, b)}$.
\end{proposition}
\begin{proof}[forthel]
  Let us show that $\Singleton{a} \Prod \Singleton{b} \Subclass \Singleton{(a, b)}$.
    Let $c \Elem \Singleton{a} \Prod \Singleton{b}$.
    Take $a' \Elem \Singleton{a}$ and $b' \Elem \Singleton{b}$ such that $c \Eq (a', b')$.
    We have $a' \Eq a$ and $b' \Eq b$.
    Hence $c \Eq (a, b)$.
    Thus $c \Elem \Singleton{(a, b)}$.
  End.

  Let us show that $\Singleton{(a, b)} \Subclass \Singleton{a} \Prod \Singleton{b}$.
    Let $c \Elem \Singleton{(a, b)}$.
    Then $c \Eq (a, b)$.
    We have $a \Elem \Singleton{a}$ and $b \Elem \Singleton{b}$.
    Hence $c \Elem \Singleton{a} \Prod \Singleton{b}$.
  End.
\end{proof}

\begin{proposition}[forthel]
  Let $A, A', B, B'$ be nonempty classes.
  If $A \Prod B \Eq A' \Prod B'$ then $A \Eq A'$ and $B \Eq B'$.
\end{proposition}
\begin{proof}[forthel]
  Assume $A \Prod B \Eq A' \Prod B'$.

  (1) $A \Subclass A'$ and $B \Subclass B'$.
  \begin{proof}
    Let $a \Elem A$ and $b \Elem B$.
    Then $(a,b) \Elem A \Prod B$.
    Hence $(a,b) \Elem A' \Prod B'$.
    Thus $a \Elem A'$ and $b \Elem B'$.
  \end{proof}

  (2) $A' \Subclass A$ and $B' \Subclass B$.
  \begin{proof}
    Let $a \Elem A'$ and $b \Elem B'$.
    Then $(a,b) \Elem A' \Prod B'$.
    Hence $(a,b) \Elem A \Prod B$.
    Thus $a \Elem A$ and $b \Elem B$.
  \end{proof}
\end{proof}
