\begin{definition*}[forthel,for=value]
  Let $f$ be a map.
  A \emph{value of $f$} is an object $b$ such that $b \Eq f(a)$ for some $a \Elem \Dom(f)$.
\end{definition*}

\begin{definition*}[forthel,for={range}]
  Let $f$ be a map.
  $\emph{\Range(f)} \DefEq \CClass{f(a)}{a \Elem \Dom(f)}$.
  Let the \emph{range of $f$} stand for $\Range(f)$.
\end{definition*}

\begin{proposition*}[forthel]
  Let $f$ be a map and $b$ be an object.
  $b$ is a value of $f$ iff $b \Elem \Range(f)$.
\end{proposition*}
\begin{proof}[forthel]
  \begin{case}{$b$ is a value of $f$.}
    Take $a \Elem \Dom(f)$ such that $b \Eq f(a)$.
    $b$ is an object.
    Hence $b \Elem \Range(f)$.
  \end{case}

  \begin{case}{$b \Elem \Range(f)$.}
    Then $b$ is an object such that $b \Eq f(a)$ for some $a \Elem \Dom(f)$.
    Hence $b$ is a value of $f$.
  \end{case}
\end{proof}

\begin{definition*}[forthel,for={identity map}]
  Let $A$ be a class.
  $\emph{\Id{A}}$ is the map defined on $A$ such that $\Id{A}(a) \Eq a$ for all $a \Elem A$.
  Let the \emph{identity map on $A$} stand for $\Id{A}$.
\end{definition*}

\begin{definition*}[forthel,for=composition]
  Let $f, g$ be maps.
  Assume $\Range(f) \Subclass \Dom(g)$.
  $\emph{g \Comp f}$ is the map defined on $\Dom(f)$ such that $(g \Comp f)(a) \Eq g(f(a))$ for all $a \Elem \Dom(f)$.
  Let the \emph{composition of $g$ and $f$} stand for $g \Comp f$.
\end{definition*}

\begin{definition*}[forthel,for=restriction]
  Let $f$ be a map and $X \Subclass \Dom(f)$.
  $\emph{f \Rest X}$ is the map defined on $X$ such that $(f \Rest X)(a) \Eq f(a)$ for all $a \Elem X$.
  Let the \emph{restriction of $f$ to $X$} stand for $f \Rest X$.
\end{definition*}

\begin{proposition*}[forthel]
  Let $A$ be a class and $X \Subclass A$.
  Then $\Id{A} \Rest X \Eq \Id{X}$.
\end{proposition*}

\begin{definition*}[forthel,for={Im,image,direct image}]
  Let $f$ be a map and $A$ be a class.
  $\emph{\Im{f}{A}} \DefEq \CClass{f(a)}{a \Elem \Dom(f) \Intersect A}$.
  Let the \emph{image of $A$ under $f$} stand for $\Im{f}{A}$.
  Let the \emph{direct image of $A$ under $f$} stand for $\Im{f}{A}$.
\end{definition*}

\begin{definition*}[forthel,for={preimage,inverse image}]
  Let $f$ be a map and $B$ be a class.
  $\emph{\PreIm{f}{B}} \DefEq \SClass{a}{\Dom(f)}{f(a) \Elem B}$.
  Let the \emph{preimage of $B$ under $f$} stand for $\PreIm{f}{B}$.
  Let the \emph{inverse image of $B$ under $f$} stand for $\PreIm{f}{B}$.
\end{definition*}

\begin{definition*}[forthel,for={map,function}]
  Let $A$ be a class.
  A \emph{map of $A$} is a map $f$ such that $\Dom(f) \Eq A$.
  Let a \emph{function of $A$} stand for a map of $A$ that is a function.
\end{definition*}

\begin{definition*}[forthel,for={map,function}]
  Let $B$ be a class.
  A \emph{map to $B$} is a map $f$ such that $f(a) \Elem B$ for each $a \Elem \Dom(f)$.
  Let a \emph{function to $B$} stand for a map to $B$ that is a function.
\end{definition*}

\begin{definition*}[forthel,for={map,FromTo,function}]
  Let $A, B$ be classes.
  A \emph{map from $A$ to $B$} is a map $f$ such that $\Dom(f) \Eq A$ and $f(a) \Elem B$ for each $a \Elem A$.
  Let $\emph{f \FromTo{A}{B}}$ stand for $f$ is a map from $A$ to $B$.
  Let a \emph{function from $A$ to $B$} stand for a map from $A$ to $B$ that is a function.
\end{definition*}

\begin{definition*}[forthel,for={map,function}]
  Let $A$ be a class.
  A \emph{map on $A$} is a map from $A$ to $A$.
  Let a \emph{function on $A$} stand for a map on $A$ that is a function.
\end{definition*}

\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f, g \FromTo{A}{B}$.
  Assume that $f(a) \Eq g(a)$ for all $a \Elem A$.
  Then $f \Eq g$.
\end{proposition*}

\begin{proposition*}
  Let $A, B$ be classes and $f$ be a map of $A$.
  Assume that $f(a) \Elem B$ for all $a \Elem A$.
  Then $f$ is a map from $A$ to $B$ iff $\Range(f) \Subclass B$.
\end{proposition*}

\begin{proposition*}[forthel]
  Let $A$ be a class.
  Then $\Id{A}$ is a map on $A$.
\end{proposition*}

\begin{proposition*}[forthel]
  Let $A, B, C$ be classes and $f \FromTo{A}{B}$ and $g \FromTo{B}{C}$.
  Then $g \Comp f \FromTo{A}{C}$.
\end{proposition*}

\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f \FromTo{A}{B}$ and $X \Subclass A$.
  Then $f \Rest X \FromTo{X}{B}$.
\end{proposition*}

\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f \FromTo{A}{B}$.
  Then $f \Comp \Id{A} \Eq f \Eq \Id{B} \Comp f$.
\end{proposition*}
\begin{proof}[forthel]
  $A$ is the domain of $f \Comp \Id{A}$ and the domain of $f$ and the domain of $\Id{B} \Comp f$.
  We have $(f \Comp \Id{A})(a)
    \Eq f(\Id{A}(a))
    \Eq f(a)
    \Eq \Id{B}(f(a))
    \Eq (\Id{B} \Comp f)(a)$
  for all $a \Elem A$.
  Hence $f \Comp \Id{A}
    \Eq f
    \Eq \Id{B} \Comp f$.
\end{proof}

\begin{proposition*}[forthel]
  Let $A$ be a class and $X \Subclass A$.
  Then $\Id{A} \Rest X \Eq \Id{X}$.
\end{proposition*}
\begin{proof}[forthel]
  We have $\Dom(\Id{A} \Rest X)
    \Eq X
    \Eq \Dom(\Id{X})$.
  $(\Id{A} \Rest X)(a)
    \Eq \Id{A}(a)
    \Eq a
    \Eq \Id{X}(a)$
  for all $a \Elem X$.
  Hence $\Id{A} \Rest X \Eq \Id{X}$.
\end{proof}

\begin{proposition*}[forthel]
  Let $A, B, C, D$ be classes and $f \FromTo{A}{B}$ and $g \FromTo{B}{C}$ and $h \FromTo{C}{D}$.
  Then $h \Comp (g \Comp f) \Eq (h \Comp g) \Comp f$.
\end{proposition*}
\begin{proof}[forthel]
  $h \Comp (g \Comp f)$ and $(h \Comp g) \Comp f$ are maps from $A$ to $D$.

  Let us show that $(h \Comp (g \Comp f))(a) \Eq ((h \Comp g) \Comp f)(a)$ for all $a \Elem A$.
    Let $a \Elem A$.
    Then $(h \Comp (g \Comp f))(a)
      \Eq h((g \Comp f)(a))
      \Eq h(g(f(a)))
      \Eq (h \Comp g)(f(a))
      \Eq ((h \Comp g) \Comp f)(a)$.
  End.

  Hence $h \Comp (g \Comp f) \Eq (h \Comp g) \Comp f$.
\end{proof}

\begin{definition*}[forthel,for=FunSpace]
  Let $A, B$ be classes.
  $\emph{\FunSpace{A}{B}}$ is the class of all functions from $A$ to $B$.
\end{definition*}

\begin{definition*}[forthel,for=fixed point]
  Let $f$ be a map.
  A \emph{fixed point of $f$} is an element $x$ of $\Dom(f)$ such that $f(x) \Eq x$.
\end{definition*}
