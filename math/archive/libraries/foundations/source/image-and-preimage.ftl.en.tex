\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f \FromTo{A}{B}$ and $X \Subclass A$.
  Then $\Im{f}{X} \Eq \CClass{f(x)}{x \Elem X}$.
\end{proposition*}

\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f \FromTo{A}{B}$.
  Then $\Im{f}{A} \Eq \Range(f)$.
\end{proposition*}

\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f \FromTo{A}{B}$ and $X \Subclass A$.
  Then $\Im{f}{X} \Eq \Range(f \Rest X)$.
\end{proposition*}

\begin{proposition*}[forthel]

  Then $\Im{\Id{A}}{X} \Eq X$.
\end{proposition*}

\begin{proposition*}[forthel]
  Let $B$ be a class and $Y \Subclass B$.
  Then $\PreIm{\Id{B}}{Y} \Eq Y$.
\end{proposition*}

\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f \FromTo{A}{B}$ and $Y \Subclass B$ and $a \Elem A$.
  Then $f(a) \Elem Y$ iff $a \Elem \PreIm{f}{Y}$.
\end{proposition*}
\begin{proof}[forthel]
  We have $\PreIm{f}{Y} \Eq \SClass{x}{A}{f(x) \Elem Y}$.
  Hence $a \Elem \PreIm{f}{Y}$ iff $a \Elem A$ and $f(a) \Elem Y$.
  Then we have the thesis.
\end{proof}

\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f \FromTo{A}{B}$.
  Then $\Im{f}{A} \Subclass B$.
\end{proposition*}
\begin{proof}[forthel]
  $\Im{f}{A}
    \Eq \Im{f}{\Dom(f)}
    \Eq \Range(f)
    \Subclass B$.
\end{proof}

\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f \FromTo{A}{B}$.
  Then $\PreIm{f}{B} \Eq A$.
\end{proposition*}
\begin{proof}[forthel]
  We have $\PreIm{f}{B} \Eq \SClass{a}{A}{f(a) \Elem B}$.
  $f(a) \Elem B$ for all $a \Elem A$.
  Hence the thesis.
\end{proof}

\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f \FromTo{A}{B}$.
  Then $\Im{f}{\PreIm{f}{B}} \Eq \Im{f}{A}$.
\end{proposition*}
\begin{proof}[forthel]
  Let us show that $\Im{f}{\PreIm{f}{B}} \Subclass \Im{f}{A}$.
    Let $b \Elem \Im{f}{\PreIm{f}{B}}$.
    Take $a \Elem \PreIm{f}{B} \Intersect A$ such that $b \Eq f(a)$.
    Then $a \Elem A$.
    Hence $b \Elem \Im{f}{A}$.
  End.

  Let us show that $\Im{f}{A} \Subclass \Im{f}{\PreIm{f}{B}}$.
    Let $b \Elem \Im{f}{A}$.
    Take $a \Elem A$ such that $b \Eq f(a)$.
    We have $b \Elem B$.
    Hence $a \Elem \PreIm{f}{B}$.
    Thus $f(a) \Elem \Im{f}{\PreIm{f}{B}}$.
    Indeed $\PreIm{f}{B} \Subclass A$.
    Therefore $b \Elem \Im{f}{\PreIm{f}{B}}$.
  End.
\end{proof}

\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f \FromTo{A}{B}$.
  Then $\PreIm{f}{\Im{f}{A}} \Eq A$.
\end{proposition*}
\begin{proof}[forthel]
  $\PreIm{f}{\Im{f}{A}} \Eq \SClass{a}{A}{f(a) \Elem \Im{f}{A}}$.
  For all $a \Elem A$ we have $f(a) \Elem \Im{f}{A}$.
  Hence every element of $\PreIm{f}{\Im{f}{A}}$ is contained in $A$ and every element of $A$ is contained in $\PreIm{f}{\Im{f}{A}}$.
  Thus $\PreIm{f}{\Im{f}{A}} \Eq A$.
\end{proof}

\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f \FromTo{A}{B}$ and $Y \Subclass B$.
  Then $\Im{f}{\PreIm{f}{Y}} \Eq Y \Intersect \Im{f}{A}$.
\end{proposition*}
\begin{proof}[forthel]
  Let us show that $\Im{f}{\PreIm{f}{Y}} \Subclass Y \Intersect \Im{f}{A}$.
    Let $b \Elem \Im{f}{\PreIm{f}{Y}}$.
    Take $a \Elem \PreIm{f}{Y}$ such that $b \Eq f(a)$.
    Then $f(a) \Elem Y \Intersect \Im{f}{A}$.
    Hence we have $b \Elem Y \Intersect \Im{f}{A}$.
  End.

  Let us show that $Y \Intersect \Im{f}{A} \Subclass \Im{f}{\PreIm{f}{Y}}$.
    Let $b \Elem Y \Intersect \Im{f}{A}$.
    Take $a \Elem A$ such that $b \Eq f(a)$.
    Then $a \Elem \PreIm{f}{Y}$.
    Hence $f(a) \Elem \Im{f}{\PreIm{f}{Y}}$.
  End.
\end{proof}

\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f \FromTo{A}{B}$ and $Y \Subclass B$.
  Then $\Im{f}{\PreIm{f}{Y}} \Subclass Y$.
\end{proposition*}

\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f \FromTo{A}{B}$ and $X \Subclass A$.
  Then $\PreIm{f}{\Im{f}{X}} \Superclass X$.
\end{proposition*}
\begin{proof}[forthel]
  Let $a \Elem X$.
  Then $f(a) \Elem \Im{f}{X}$.
  Hence $a \Elem \PreIm{f}{\Im{f}{X}}$.
  Indeed $\Im{f}{X} \Subclass B$.
\end{proof}

\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f \FromTo{A}{B}$ and $X \Subclass A$.
  Then $\Im{f}{X} \Eq \EmptyClass$ iff $X \Eq \EmptyClass$.
\end{proposition*}
\begin{proof}[forthel]
  \begin{case}{$\Im{f}{X}$ is empty.}
    Then there is no $a \Elem X$ such that $f(a) \Elem \Im{f}{X}$.
    For all $a \Elem X$ we have $f(a) \Elem \Im{f}{X}$.
    Hence $X$ is empty.
  \end{case}

  \begin{case}{$X$ is empty.}
    For all $b \Elem \Im{f}{X}$ we have $b \Eq f(a)$ for some $a \Elem X$.
    There is no $a \Elem X$.
    Hence $\Im{f}{X}$ is empty.
  \end{case}
\end{proof}

\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f \FromTo{A}{B}$ and $Y \Subclass B$.
  Then $\PreIm{f}{Y} \Eq \EmptyClass$ iff $Y \Subclass B \Diff \Im{f}{A}$.
\end{proposition*}
\begin{proof}[forthel]
  \begin{case}{$\PreIm{f}{Y}$ is empty.}
    Let $b \Elem Y$.
    Then $b \Elem B$.

    There is no $a \Elem A$ such that $b \Eq f(a)$.
    \begin{proof}
      Assume the contrary.
      Take $a \Elem A$ such that $b \Eq f(a)$.
      Then $a \Elem \PreIm{f}{Y}$.
      Contradiction.
    \end{proof}

    Hence $b \NotElem \Im{f}{A}$.
    Therefore $b \Elem B \Diff \Im{f}{A}$.
  \end{case}

  \begin{case}{$Y \Subclass B \Diff \Im{f}{A}$.}
    Then no element of $Y$ is an element of $\Im{f}{A}$.
    Assume that $\PreIm{f}{Y}$ is nonempty.
    Take $a \Elem \PreIm{f}{Y}$.
    Then $f(a) \Elem Y$ and $f(a) \Elem \Im{f}{A}$.
    Contradiction.
  \end{case}
\end{proof}

\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f \FromTo{A}{B}$ and $X \Subclass A$ and $Y \Subclass B$.
  Then $\Im{f}{X} \Intersect Y \Eq \EmptyClass$ iff $X \Intersect \PreIm{f}{Y} \Eq \EmptyClass$.
\end{proposition*}
\begin{proof}[forthel]
  \begin{case}{$\Im{f}{X} \Intersect Y$ is empty.}
    Assume that $X \Intersect \PreIm{f}{Y}$ is nonempty.
    Take $a \Elem X \Intersect \PreIm{f}{Y}$.
    Then $f(a) \Elem \Im{f}{X}$ and $f(a) \Elem Y$.
    Hence $f(a) \Elem \Im{f}{X} \Intersect Y$.
    Contradiction.
  \end{case}

  \begin{case}{$X \Intersect \PreIm{f}{Y}$ is empty.}
    Assume that $\Im{f}{X} \Intersect Y$ is nonempty.
    Take $b \Elem \Im{f}{X} \Intersect Y$.
    Consider a $a \Elem X$ such that $b \Eq f(a)$.
    Then $a \Elem \PreIm{f}{Y}$.
    Indeed $b \Elem Y$.
    Hence $a \Elem X \Intersect \PreIm{f}{Y}$.
    Contradiction.
  \end{case}
\end{proof}

\begin{proposition*}[forthel]
  Let $A, B, C$ be classes and $f \FromTo{A}{B}$ and $g \FromTo{B}{C}$ and $X \Subclass A$.
  Then $\Im{g \Comp f}{X} \Eq \Im{g}{\Im{f}{X}}$.
\end{proposition*}
\begin{proof}[forthel]
  $\Im{g \Comp f}{X} \Eq \CClass{g(f(a))}{a \Elem X}$.
  $\Im{f}{X}$ is a subclass of $B$.
  We have $\Im{g}{\Im{f}{X}} \Eq \CClass{g(b)}{b \Elem \Im{f}{X}}\text{ and }\Im{f}{X} \Eq \CClass{f(a)}{a \Elem X}$.
  Thus $\Im{g}{\Im{f}{X}} \Eq \CClass{g(f(a))}{a \Elem X}$.
  Therefore $\Im{g \Comp f}{X} \Eq \Im{g}{\Im{f}{X}}$.
\end{proof}

\begin{proposition*}[forthel]
  Let $A, B, C$ be classes and $f \FromTo{A}{B}$ and $g \FromTo{B}{C}$ and $Z \Subclass C$.
  Then $\PreIm{g \Comp f}{Z} \Eq \PreIm{f}{\PreIm{g}{Z}}$.
\end{proposition*}
\begin{proof}[forthel]
  $\PreIm{g \Comp f}{Z} \Eq \SClass{a}{A}{g(f(a)) \Elem Z}$.
  We have $\PreIm{g}{Z} \Eq \SClass{b}{B}{g(b) \Elem Z}$ and $\PreIm{f}{\PreIm{g}{Z}} \Eq \SClass{a}{A}{f(a) \Elem \PreIm{g}{Z}}$.
  Hence $\PreIm{f}{\PreIm{g}{Z}} \Eq \SClass{a}{A}{g(f(a)) \Elem Z}$.
  Thus $\PreIm{g \Comp f}{Z} \Eq \PreIm{f}{\PreIm{g}{Z}}$.
\end{proof}

\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f \FromTo{A}{B}$ and $X, X' \Subclass A$.
  If $X \Subclass X'$ then $\Im{f}{X} \Subclass \Im{f}{X'}$.
\end{proposition*}
\begin{proof}[forthel]
  Assume $X \Subclass X'$.
  Let $b \Elem \Im{f}{X}$.
  Take $a \Elem X$ such that $f(a) \Eq b$.
  Then $a \Elem X'$.
  Hence $b \Eq f(a) \Elem \Im{f}{X'}$.
\end{proof}

\begin{proposition*}[forthel]

\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f \FromTo{A}{B}$ and $X, X' \Subclass A$.
  Then $\Im{f}{X \Union X'} \Eq \Im{f}{X} \Union \Im{f}{X'}$.
\end{proposition*}
\begin{proof}[forthel]
  Let us show that $\Im{f}{X \Union X'} \Subclass \Im{f}{X} \Union \Im{f}{X'}$.
    Let $b \Elem \Im{f}{X \Union X'}$.
    Take $a \Elem X \Union X'$ such that $b \Eq f(a)$.
    Then $a \Elem X$ or $a \Elem X'$.
    Hence $f(a) \Elem \Im{f}{X}$ or $f(a) \Elem \Im{f}{X'}$.
    Thus $b \Eq f(a) \Elem \Im{f}{X} \Union \Im{f}{X'}$.
  End.

  Let us show that $\Im{f}{X} \Union \Im{f}{X'} \Subclass \Im{f}{X \Union X'}$.
    Let $b \Elem \Im{f}{X} \Union \Im{f}{X'}$.

    \begin{case}{$b \Elem \Im{f}{X}$.}
      Take $a \Elem X$ such that $b \Eq f(a)$.
      Then $a \Elem X \Union X'$.
      Hence $b \Elem \Im{f}{X \Union X'}$.
    \end{case}

    \begin{case}{$b \Elem \Im{f}{X'}$.}
      Take $a \Elem X'$ such that $b \Eq f(a)$.
      Then $a \Elem X \Union X'$.
      Hence $b \Elem \Im{f}{X \Union X'}$.
    \end{case}
  End.
\end{proof}

\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f \FromTo{A}{B}$ and $Y, Y' \Subclass B$.
  Then $\PreIm{f}{Y \Union Y'} \Eq \PreIm{f}{Y} \Union \PreIm{f}{Y'}$.
\end{proposition*}
\begin{proof}[forthel]
  Let us show that $\PreIm{f}{Y \Union Y'} \Subclass \PreIm{f}{Y} \Union \PreIm{f}{Y'}$.
    Let $a \Elem \PreIm{f}{Y \Union Y'}$.
    Then $f(a) \Elem Y \Union Y'$.
    Hence $f(a) \Elem Y$ or $f(a) \Elem Y'$.
    If $f(a) \Elem Y$ then $a \Elem \PreIm{f}{Y}$.
    If $f(a) \Elem Y'$ then $a \Elem \PreIm{f}{Y'}$.
    Thus $a \Elem \PreIm{f}{Y} \Union \PreIm{f}{Y'}$.
  End.

  Let us show that $\PreIm{f}{Y} \Union \PreIm{f}{Y'} \Subclass \PreIm{f}{Y \Union Y'}$.
    Let $a \Elem \PreIm{f}{Y} \Union \PreIm{f}{Y'}$.
    Then $a \Elem \PreIm{f}{Y}$ or $a \Elem \PreIm{f}{Y'}$.
    If $a \Elem \PreIm{f}{Y}$ then $f(a) \Elem Y$.
    If $a \Elem \PreIm{f}{Y'}$ then $f(a) \Elem Y'$.
    Hence $f(a) \Elem Y \Union Y'$.
    Thus $a \Elem \PreIm{f}{Y \Union Y'}$.
  End.
\end{proof}

\begin{proposition*}[forthel]
  Let $A, B$ be classes and $f \FromTo{A}{B}$ and $X, X' \Subclass A$.
  Then $\Im{f}{X \Diff X'} \Superclass \Im{f}{X} \Diff \Im{f}{X'}$.
\end{proposition*}
\begin{proof}[forthel]
  Let $b \Elem \Im{f}{X} \Diff \Im{f}{X'}$.
  Then $b \Elem \Im{f}{X}$ and $b \NotElem \Im{f}{X'}$.
  Take $a \Elem X$ such that $b \Eq f(a)$.
  If $a \Elem X'$ then $b \Elem \Im{f}{X'}$.
  Hence $a \NotElem X'$.
  Thus $a \Elem X \Diff X'$.
  Therefore $b \Eq f(a) \Elem \Im{f}{X \Diff X'}$.
\end{proof}

  End.
\end{proof}
