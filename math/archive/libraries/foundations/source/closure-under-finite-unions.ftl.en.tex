\begin{smodule}{closure-under-finite-unions.ftl}

\importmodule[libraries/foundations]{systems-of-sets.ftl}
\importmodule[libraries/foundations]{finite-and-infinite-classes.ftl}

\symdecl*{closed under finite unions}

\begin{definition*}[forthel,for=closed under finite unions]
  Let $X$ be a system of sets.
  $X$ is \emph{closed under finite unions} iff $\UnionOver U \Elem X$ for every nonempty finite subclass $U$ of $X$.
\end{definition*}

\begin{proposition*}[forthel]
  Let $X$ be a system of sets.
  $X$ is closed under finite unions iff $U \Union V \Elem X$ for every $U, V \Elem X$.
\end{proposition*}
\begin{proof}[forthel]
  \begin{case}{$X$ is closed under finite unions.}
    Let $U, V \Elem X$.
    Then $\Pair{U}{V}$ is a nonempty finite subclass of $X$.
    Hence $U \Union V \Eq \UnionOver \Pair{U}{V} \Elem X$.
  \end{case}

  \begin{case}{$U \Union V \Elem X$ for every $U, V \Elem X$.}
    Define $\Phi \DefEq \SClass{n}{\Naturals}{\UnionOver U \Elem X\text{ for every nonempty subclass }U\text{ of }X\text{ that has }n\text{ elements}}$.

    (1) $\Phi$ contains $\Zero$.

    (2) For every $n \Elem \Phi$ we have $n \Plus \One \Elem \Phi$.
    \begin{proof}
      Let $n \Elem \Phi$.
      Then $\UnionOver U \Elem X$ for every nonempty subclass $U$ of $X$ that has $n$ elements.

      Let us show that $\UnionOver U \Elem X$ for every nonempty subclass $U$ of $X$ that has $n \Plus \One$ elements.

        \begin{case}{$n \Eq \Zero$.} \end{case}

        \begin{case}{$n \NotEq \Zero$.}
          Let $U$ be a nonempty subclass of $X$ such that $U$ has $n \Plus \One$ elements.
          Take a bijection $f$ between $\NatSegment{\One}{n \Plus \One}$ and $U$.
          We have $\NatSegment{\One}{n \Plus \One} \Eq \NatSegment{\One}{n} \Union \Singleton{n \Plus \One}$.
          Take $V \Eq \Im{f}{\NatSegment{\One}{n}}$.
          We have $\NatSegment{\One}{n} \Subclass \NatSegment{\One}{n \Plus \One}$.

          Let us show that $V \Subclass U$.
            Let $x \Elem V$.
            Take $k \Elem \NatSegment{\One}{n}$ such that $x \Eq f(k)$.
            Indeed we can show that there exists a $k \Elem \NatSegment{\One}{n}$ such that $x \Eq f(k)$.
              Assume the contrary.
              Then $x \NotEq f(k)$ for all $k \Elem \NatSegment{\One}{n}$.
              Hence $x \NotElem \Im{f}{\NatSegment{\One}{n}} \Eq V$.
              Contradiction.
            End.
            Hence $x \Elem U$.
            Indeed $x \Elem \Im{f}{\NatSegment{\One}{n \Plus \One}}$.
          End.

          $V$ is nonempty.
          Indeed $f(\One) \Elem \Im{f}{\NatSegment{\One}{n}}$.
          Indeed $\One \Elem \NatSegment{\One}{n}$.
          Hence $f \Rest \NatSegment{\One}{n}$ is a bijection between $\NatSegment{\One}{n}$ and $V$ (by \sn{bijectivity of restriction of injection}).
          Thus $V$ has $n$ elements.
          Consequently $\UnionOver V \Elem X$.

          Let us show that $U \Eq V \Union \Singleton{f(n \Plus \One)}$.

            (1) $\Im{f}{A \Union B} \Eq \Im{f}{A} \Union \Im{f}{B}$ for all $A, B \Subclass \Dom(f)$.

            (2) $\Im{f}{\Singleton{a}} \Eq \Singleton{f(a)}$ for all $a \Elem \Dom(f)$.

            Hence $U
              \Eq \Im{f}{\Dom(f)}
              \Eq \Im{f}{\NatSegment{\One}{n \Plus \One}}
              \Eq \Im{f}{\NatSegment{\One}{n} \Union \Singleton{n \Plus \One}}
              \Eq \Im{f}{\NatSegment{\One}{n}} \Union \Im{f}{\Singleton{n \Plus \One}}
              \Eq \Im{f}{\NatSegment{\One}{n}} \Union \Singleton{f(n \Plus \One)}
              \Eq V \Union \Singleton{f(n \Plus \One)}$.
            Indeed $n \Plus \One \Elem \Dom(f)$ and $\NatSegment{\One}{n}, \Singleton{\One \Plus n} \Subclass \Dom(f)$.
          End.

          Let us show that $\UnionOver (A \Union B) \Eq (\UnionOver A) \Union (\UnionOver B)$ for any nonempty systems of sets $A, B$.
            Let $A, B$ be nonempty systems of sets.
            $\UnionOver (A \Union B) \Subclass (\UnionOver A) \Union (\UnionOver B)$.
            $((\UnionOver A) \Union (\UnionOver B)) \Subclass \UnionOver (A \Union B)$. %!
          End.

          Let us show that $f(n \Plus \One)$ and $f(k)$ are sets for every $k \Elem \NatSegment{\One}{n}$.
            Let $k \Elem \NatSegment{\One}{n}$.
            Then $f(k), f(n \Plus \One) \Elem U$.
            Hence $f(k)$ and $f(n \Plus \One)$ are sets.
          End.

          Hence $V$ and $\Singleton{f(n \Plus \One)}$ are nonempty systems of sets.
          Thus $\UnionOver U
            \Eq \UnionOver (V \Union \Singleton{f(n \Plus \One)})
            \Eq (\UnionOver V) \Union (\UnionOver \Singleton{f(n \Plus \One)})
            \Eq (\UnionOver V) \Union f(n \Plus \One)
            \Elem X$.
        \end{case}
      End.
    \end{proof}

    Therefore $\Phi$ contains every natural number (by \sr{induction I}{induction}).
    Thus $\UnionOver U \Elem X$ for every nonempty finite subclass $U$ of $X$.
    Consequently $X$ is closed under finite unions.
  \end{case}
\end{proof}
