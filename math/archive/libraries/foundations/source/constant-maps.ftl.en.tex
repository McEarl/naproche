\begin{smodule}{constant-maps.ftl}

\importmodule[libraries/foundations]{injections-surjections-bijections.ftl}

\symdef{Const}[args=2]{\comp{\textrm{const}}_{#1}^{#2}}

\symdecl*{constant}

\begin{definition*}[forthel,for=constant]
  Let $f$ be a map.
  $f$ is \emph{constant} iff there exists an object $a$ such that $f(x) \Eq a$ for all $x \Elem \Dom(f)$.
\end{definition*}

\begin{definition*}[forthel,for=Const]
  Let $A$ be a class and $a$ be an object.
  $\emph{\Const{A}{a}}$ is the map $f$ such that $\Dom(f) \Eq A$ and $f(x) \Eq a$ for all $x \Elem A$.
\end{definition*}

\begin{proposition*}[forthel]
  Let $A$ be a nonempty class and $a$ be an object.
  $\Const{A}{a}$ is a surjective map from $A$ onto $\Singleton{a}$.
\end{proposition*}
