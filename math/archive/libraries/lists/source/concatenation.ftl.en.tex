\begin{smodule}{concatenation.ftl}
\importmodule[libraries/lists]{lists.ftl}
\symdef{Concat}[op=+\mkern-10mu+]{\,\comp+\mkern-10mu\comp+\,}

\begin{signature}[forthel,for=Concat]
  Let $L, L'$ be lists.
  $\emph{L \Concat L'}$ is a list.
\end{signature}

\begin{axiom}[forthel,for=Concat]
  Let $L$ be a list.
  Then $\List{} \Concat L \Eq L$.
\end{axiom}

\begin{axiom}[forthel,for=Concat]
  Let $a$ be an object and $L, L'$ be lists.
  Then $(a \Cons L) \Concat L' \Eq a \Cons (L \Concat L')$.
\end{axiom}

\begin{proposition}[forthel]
  $L \Concat \List{} \Eq L$ for every list $L$.
\end{proposition}
\begin{proof}[forthel,method=induction on $L$]
  Let $L$ be a list.

  \begin{case}{$L \Eq \List{}$.} \end{case}
  
  \begin{case}{$L \Eq a \Cons L'$ for some object $a$ and some list $L'$.}
    Consider an object $a$ and a list $L'$ such that $L \Eq a \Cons L'$.
    Then $L' \ILess L$.
    Hence $L' \Concat \List{} \Eq L'$.
    Thus
    \begin{align*}
      L \Concat \List{}
        &\Eq (a \Cons L') \Concat \List{}\\
        &\Eq a \Cons (L' \Concat \List{})\\
        &\Eq a \Cons L'\\
        &\Eq L
    \end{align*}.
  \end{case}
\end{proof}

\symdecl*{associativity of concatenation}

\begin{proposition}[forthel,name=associativity of concatenation]
  $L \Concat (L' \Concat L'') \Eq (L \Concat L') \Concat L''$ for any lists $L, L', L''$.
\end{proposition}
\begin{proof}[forthel,method=induction on $L$]
  Let $L$ be a list.

  \begin{case}{$L \Eq \List{}$.} \end{case}

  \begin{case}{$L \Eq a \Cons L'''$ for some object $a$ and some list $L'''$.}
    Consider an object $a$ and a list $L'''$ such that $L \Eq a \Cons L'''$.
    Then $L''' \ILess L$.
    Let $L',L''$ be lists.
    Then $L''' \Concat (L' \Concat L'') \Eq (L''' \Concat L') \Concat L''$.
    Thus
    \begin{align*}
      L \Concat (L' \Concat L'') 
        &\Eq (a \Cons L''') \Concat (L' \Concat L'')\\
        &\Eq a \Cons (L''' \Concat (L' \Concat L''))\\
        &\Eq a \Cons ((L''' \Concat L') \Concat L'')\\
        &\Eq (a \Cons (L''' \Concat L')) \Concat L''\\
        &\Eq ((a \Cons L''') \Concat L') \Concat L''\\
        &\Eq (L \Concat L') \Concat L''
    \end{align*}.
  \end{case}
\end{proof}
