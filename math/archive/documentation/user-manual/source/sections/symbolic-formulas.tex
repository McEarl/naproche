\begin{sfragment}[id=sec:symbolic-formulas]{Symbolic Formulas}
  \begin{sparagraph}[style=symdoc]
    \vardef{phivar}{\varphi}
    \vardef{psivar}{\psi}
    \vardef{xvar}{x}
    \vardef{nuvar}{\nu}
    \vardef{Svar}{S}
    \vardef{Rvar}{R}
    \vardef{xvar}{x}
    %
    \Definame[post=s]{symbolic formula} are recursively defined as follows:
    \begin{enumerate}
      \item $\defnotation\TRUE$ is a \sn{symbolic formula};
      \item $\defnotation\FALSE$ is a \sn{symbolic formula};
      \item if $\phivar$ is a \sn{symbolic formula},
        then $\defnotation\NOT\phivar$ is a \sn{symbolic formula};
      \item if $\phivar$ and $\psivar$ are \sn{symbolic formula},
        then $\phivar\defnotation\AND\psivar$ is a \sn{symbolic formula};
      \item if $\phivar$ and $\psivar$ are \sns{symbolic formula},
        then $\phivar\defnotation\OR\psivar$ is a \sn{symbolic formula};
      \item if $\phivar$ and $\psivar$ are \sns{symbolic formula},
        then $\phivar\defnotation\IMPLIES\psivar$ is a \sn{symbolic formula};
      \item if $\phivar$ and $\psivar$ are \sns{symbolic formula},
        then $\phivar\defnotation\IFF\psivar$ is a \sn{symbolic formula};
      \item if $\nuvar$ a symbolic notion and $\phivar$ a \sn{symbolic formula},
        then $\defnotation\FORALL\nu\phivar$ is a \sn{symbolic formula};
      \item if $\nuvar$ a symbolic notion and $\phivar$ a \sn{symbolic formula},
        then $\defnotation\EXISTS\nuvar\phivar$ is a \sn{symbolic formula};
      \item if $\Svar$ is a statement,
        then $(\Svar)$ is a \sn{symbolic formula};
      \item if $\Rvar$ is a symbolic relation,
        then $\Rvar$ is a \sn{symbolic formula}.
    \end{enumerate}

    \noindent
    The \sn{first-order semantics} of \sns{symbolic formula} is given as follows:
    \begin{enumerate}
      \item $\folsem\TRUE=\foltrue$;
      \item $\folsem\FALSE=\folfalse$;
      \item $\folsem{\NOT\phivar}=\folneg{\folsem\phivar}$;
      \item $\folsem{\phivar\AND\psivar}=\folconj{\folsem\phivar,\folsem\psivar}$;
      \item $\folsem{\phivar\OR\psivar}=\foldisj{\folsem\phivar,\folsem\psivar}$;
      \item $\folsem{\phivar\IMPLIES\psivar}=\folimpl{\folsem\phivar}{\folsem\psivar}$;
      \item $\folsem{\phivar\IFF\psivar}=\folequiv{\folsem\phivar,\folsem\psivar}$;
      \item $\folsem{\FORALL\nuvar\phivar}=\folforall\xvar{\folimpl{\folsem\nuvar}{\folsem\phivar}}$,
        where $\xvar$ is the variable declaration given by $\nuvar$;
      \item $\folsem{\EXISTS\nuvar\phivar}=\folexists\xvar{\folconj{\folsem\nuvar,\folsem\phivar}}$,
        where $\xvar$ is the variable declaration given by $\nuvar$;
      \item $\folsem{(\Svar)}=\folsem\Svar$.
    \end{enumerate}

    \Sns{symbolic formula} can be used in a \ForTheL text via the notation given in the
    table below:

    \begin{center}
      \begin{tabular}{|l||l|l|l|}
        \hline
        & ASCII Dialect & \TeX Dialect & \sTeX Dialect \\\hline\hline
        $\TRUE$ &
          \texttt{true} &
          \texttt{\char`\\top} &
          \texttt{\char`\\TRUE} \\\hline
        $\FALSE$ &
          \texttt{false} &
          \texttt{\char`\\bot} & \texttt{\char`\\FALSE} \\\hline
        $\NOT\phivar$ &
          \texttt{not :} $\phivar$ &
          \texttt{\char`\\neg} $\phivar$
           & \texttt{\char`\\NOT} $\phivar$ \\\hline
        $\phivar\AND\psivar$ &
          $\phivar$ \texttt{/\char`\\} $\psivar$ &
          $\phivar$ \texttt{\char`\\wedge} $\psivar$ &
          $\phivar$ \texttt{\char`\\AND} $\psivar$ \\\hline
        $\phivar\OR\psivar$ &
          $\phivar$ \texttt{\char`\\/} $\psivar$ &
          $\phivar$ \texttt{\char`\\vee} $\psivar$ &
          $\phivar$ \texttt{\char`\\OR} $\psivar$ \\\hline
        $\phivar\IMPLIES\phivar$ &
          $\phivar$ \texttt{=>} $\psivar$ &
          $\phivar$ \texttt{\char`\\implies} $\psivar$ &
          $\phivar$ \texttt{\char`\\IMPLIES} $\psivar$ \\\hline
        $\phivar\IFF\phivar$ &
          $\phivar$ \texttt{<=>} $\psivar$ &
          $\phivar$ \texttt{\char`\\iff} $\psivar$ &
          $\phivar$ \texttt{\char`\\IFF} $\psivar$ \\\hline
        $\FORALL\nuvar\phivar$ &
          \texttt{forall} $\nuvar$ \texttt: $\phivar$ &
          \texttt{\char`\\forall} $\nuvar$ \texttt: $\phivar$ &
          \texttt{\char`\\FORALL\char`\{}$\nuvar$\texttt{\char`\}} $\phivar$ \\\hline
        $\EXISTS\nuvar\phivar$ &
          \texttt{exists} $\nuvar$ \texttt: $\phivar$ &
          \texttt{\char`\\exists} $\nuvar$ \texttt: $\phivar$ &
          \texttt{\char`\\EXISTS\char`\{}$\nuvar$\texttt{\char`\}} $\phivar$ \\\hline
      \end{tabular}
    \end{center}
  \end{sparagraph}
\end{sfragment}
