\begin{sfragment}[id=sec:symbolic-formulas]{Symbolic Formulas}
  \symdecl*{symbolic formula}
  %
  \begin{sparagraph}[style=symdoc]
    \vardef{phivar}{\varphi}
    \vardef{psivar}{\psi}
    \vardef{xvar}{x}
    \vardef{nuvar}{\nu}
    \vardef{Svar}{S}
    \vardef{Rvar}{R}
    \vardef{xvar}{x}
    %
    In the \asciidialect, \definame[post=s]{symbolic formula} are recursively defined as follows:
    \begin{enumerate}
      \item ``\texttt{true}'' is a \sn{symbolic formula};
      \item ``\texttt{false}'' is a \sn{symbolic formula};
      \item if $\phivar$ is a \sn{symbolic formula},
        then ``\texttt{not :} $\phivar$'' is a \sn{symbolic formula};
      \item if $\phivar$ and $\psivar$ are \sn{symbolic formula},
        then ``$\phivar$ \texttt{/\char`\\} $\psivar$'' is a \sn{symbolic formula};
      \item if $\phivar$ and $\psivar$ are \sns{symbolic formula},
        then ``$\phivar$ \texttt{\char`\\/} $\psivar$'' is a \sn{symbolic formula};
      \item if $\phivar$ and $\psivar$ are \sns{symbolic formula},
        then ``$\phivar$ \texttt{=>} $\psivar$'' is a \sn{symbolic formula};
      \item if $\phivar$ and $\psivar$ are \sns{symbolic formula},
        then ``$\phivar$ \texttt{<=>} $\psivar$'' is a \sn{symbolic formula};
      \item if $\nuvar$ a symbolic notion and $\phivar$ a \sn{symbolic formula},
        then ``\texttt{forall} $\nuvar$ \texttt: $\phivar$'' is a \sn{symbolic formula};
      \item if $\nuvar$ a symbolic notion and $\phivar$ a \sn{symbolic formula},
        then ``\texttt{exists} $\nuvar$ \texttt: $\phivar$'' is a \sn{symbolic formula};
      \item if $\Svar$ is a statement,
        then ``\texttt($\Svar$\texttt)'' is a \sn{symbolic formula};
      \item if $\Rvar$ is a symbolic relation,
        then ``$\Rvar$'' is a \sn{symbolic formula}.
    \end{enumerate}

    In the \texdialect, \definame[post=s]{symbolic formula} are recursively defined as follows:
    \begin{enumerate}
      \item ``\texttt{\char`\\top}'' is a \sn{symbolic formula};
      \item ``\texttt{\char`\\bot}'' is a \sn{symbolic formula};
      \item if $\phivar$ is a \sn{symbolic formula},
        then ``\texttt{\char`\\neg} $\phivar$'' is a \sn{symbolic formula};
      \item if $\phivar$ and $\psivar$ are \sn{symbolic formula},
        then ``$\phivar$ \texttt{\char`\\wedge} $\psivar$'' is a \sn{symbolic formula};
      \item if $\phivar$ and $\psivar$ are \sns{symbolic formula},
        then ``$\phivar$ \texttt{\char`\\vee} $\psivar$'' is a \sn{symbolic formula};
      \item if $\phivar$ and $\psivar$ are \sns{symbolic formula},
        then ``$\phivar$ \texttt{\char`\\implies} $\psivar$'' is a \sn{symbolic formula};
      \item if $\phivar$ and $\psivar$ are \sns{symbolic formula},
        then ``$\phivar$ \texttt{\char`\\iff} $\psivar$'' is a \sn{symbolic formula};
      \item if $\nuvar$ a symbolic notion and $\phivar$ a \sn{symbolic formula},
        then ``\texttt{\char`\\forall} $\nuvar$ \texttt: $\phivar$'' is a \sn{symbolic formula};
      \item if $\nuvar$ a symbolic notion and $\phivar$ a \sn{symbolic formula},
        then ``\texttt{\char`\\exists} $\nuvar$ \texttt: $\phivar$'' is a \sn{symbolic formula};
      \item if $\Svar$ is a statement,
        then ``\texttt($\Svar$\texttt)'' is a \sn{symbolic formula};
      \item if $\Rvar$ is a symbolic relation,
        then ``$\Rvar$'' is a \sn{symbolic formula}.
    \end{enumerate}

    In the \stexdialect, \definame[post=s]{symbolic formula} are recursively defined as follows:
    \begin{enumerate}
      \item ``\texttt{\char`\\TRUE}'' is a \sn{symbolic formula};
      \item ``\texttt{\char`\\FALSE}'' is a \sn{symbolic formula};
      \item if $\phivar$ is a \sn{symbolic formula},
        then ``\texttt{\char`\\NOT\char`\{}$\phivar$\texttt{\char`\}}'' is a \sn{symbolic formula};
      \item if $\phivar$ and $\psivar$ are \sn{symbolic formula},
        then ``\texttt{\char`\\AND\char`\{}$\phivar$\texttt{\char`\}\char`\{}$\psivar$\texttt{\char`\}}'' is a \sn{symbolic formula};
      \item if $\phivar$ and $\psivar$ are \sns{symbolic formula},
        then ``\texttt{\char`\\OR\char`\{}$\phivar$\texttt{\char`\}\char`\{}$\psivar$\texttt{\char`\}}'' is a \sn{symbolic formula};
      \item if $\phivar$ and $\psivar$ are \sns{symbolic formula},
        then ``\texttt{\char`\\IMPLIES\char`\{}$\phivar$\texttt{\char`\}\char`\{}$\psivar$\texttt{\char`\}}'' is a \sn{symbolic formula};
      \item if $\phivar$ and $\psivar$ are \sns{symbolic formula},
        then ``\texttt{\char`\\IFF\char`\{}$\phivar$\texttt{\char`\}\char`\{}$\psivar$\texttt{\char`\}}'' is a \sn{symbolic formula};
      \item if $\nuvar$ a symbolic notion and $\phivar$ a \sn{symbolic formula},
        then ``\texttt{\char`\\FORALL\char`\{}$\nuvar$\texttt{\char`\}\char`\{}$\phivar$\texttt{\char`\}}'' is a \sn{symbolic formula};
      \item if $\nuvar$ a symbolic notion and $\phivar$ a \sn{symbolic formula},
        then ``\texttt{\char`\\EXISTS\char`\{}$\nuvar$\texttt{\char`\}\char`\{}$\phivar$\texttt{\char`\}}'' is a \sn{symbolic formula};
      \item if $\Svar$ is a statement,
        then ``\texttt($\Svar$\texttt)'' is a \sn{symbolic formula};
      \item if $\Rvar$ is a symbolic relation,
        then ``$\Rvar$'' is a \sn{symbolic formula}.
    \end{enumerate}

    \noindent
    The \sn{first-order semantics} of \sns{symbolic formula} is given as follows
    (we only state them for the \asciidialect; the \sr{first-order semantics}{semantics}
    for the \sr{texdialect}{\TeX} and \stexdialect is defined analogous):
    \begin{enumerate}
      \item $\folsem{\texttt{true}}=\foltrue$;
      \item $\folsem{\texttt{false}}=\folfalse$;
      \item $\folsem{\texttt{not }\phivar}=\folneg{\folsem\phivar}$;
      \item $\folsem{\phivar\texttt{ /\char`\\\ }\psivar}=\folconj{\folsem\phivar,\folsem\psivar}$; % the non-breaking space is intended!
      \item $\folsem{\phivar\texttt{ \char`\\/ }\psivar}=\foldisj{\folsem\phivar,\folsem\psivar}$;
      \item $\folsem{\phivar\texttt{ => }\psivar}=\folimpl{\folsem\phivar}{\folsem\psivar}$;
      \item $\folsem{\phivar\texttt{ <=> }\psivar}=\folequiv{\folsem\phivar,\folsem\psivar}$;
      \item $\folsem{\texttt{forall }\nuvar\texttt{ : }\phivar}=\folforall\xvar{\folimpl{\folsem\nuvar}{\folsem\phivar}}$,
        where $\xvar$ is the variable declaration given by $\nuvar$;
      \item $\folsem{\texttt{exists }\nuvar\texttt{ : }\phivar}=\folexists\xvar{\folconj{\folsem\nuvar,\folsem\phivar}}$,
        where $\xvar$ is the variable declaration given by $\nuvar$;
      \item $\folsem{\texttt(\Svar\texttt)}=\folsem\Svar$.
    \end{enumerate}
  \end{sparagraph}
\end{sfragment}
